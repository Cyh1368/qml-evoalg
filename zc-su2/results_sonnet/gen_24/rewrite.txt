"""Seed program. Only ANSATZ_SPEC inside the EVOLVE-BLOCK is evolved.

Everything else about the task, that is how inputs are encoded, how the circuit
is measured, how training works and how metrics are computed, is fixed and lives
in a module that is not reproduced here. No information about the data is
available in this file.
"""

from _backend import run_experiment as _run

N_QUBITS = 8
ALLOWED_SINGLE_QUBIT_GATES = {"RX", "RY", "RZ"}
ALLOWED_TWO_QUBIT_GATES = {"CNOT", "CZ"}
ALLOWED_PARAM_TWO_QUBIT_GATES = {"CRX", "CRY", "CRZ"}
ALLOWED_ISING_GATES = {"XX", "YY", "ZZ"}


# EVOLVE-BLOCK-START
ANSATZ_SPEC = []

# 1. Pre-entanglement rotation layer: ALL 8 qubits share a SINGLE global
#    trainable parameter "theta" (maximal parameter sharing, taken to the
#    logical extreme of the global-RY seed, but reduced from 2 unique
#    parameters per block down to just 1).
for wire in range(N_QUBITS):
    ANSATZ_SPEC.append({"gate": "RY", "wire": wire, "param": "theta"})

# 2. Entangling layer: open nearest-neighbor CZ chain (no wrap-around edge).
#    This is the low-depth, fast-converging topology from the coarse-grained
#    RY/RZ seed (depth 29, convergence in ~30 steps), rather than the closed
#    ring used by the global-RY/RZ seed (depth 60, convergence in ~150
#    steps). Zero extra trainable parameters, full register connectivity.
for i in range(N_QUBITS - 1):
    ANSATZ_SPEC.append({"gate": "CZ", "wires": [i, i + 1]})

# 3. Post-entanglement rotation layer: reuse the SAME single shared
#    parameter "theta" (same name as step 1), on a different rotation axis
#    (RZ), so the block still realizes a non-trivial entangled
#    transformation while using only ONE unique parameter name in total.
for wire in range(N_QUBITS):
    ANSATZ_SPEC.append({"gate": "RZ", "wire": wire, "param": "theta"})
# EVOLVE-BLOCK-END


def run_experiment(**kwargs):
    return _run(ANSATZ_SPEC, **kwargs)
