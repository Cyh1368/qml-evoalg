ANSATZ_SPEC = []

# Layer 1: Full single-qubit rotation with RY + RZ
for i in range(8):
    ANSATZ_SPEC.append({"gate": "RY", "wire": i, "param": f"ry1_{i}"})
    ANSATZ_SPEC.append({"gate": "RZ", "wire": i, "param": f"rz1_{i}"})

# Circular CNOT ring: nearest-neighbor entanglement with wrap-around
for i in range(8):
    ANSATZ_SPEC.append({"gate": "CNOT", "wires": [i, (i + 1) % 8]})

# Layer 2: RY rotation between entanglement stages
for i in range(8):
    ANSATZ_SPEC.append({"gate": "RY", "wire": i, "param": f"ry2_{i}"})

# Long-range cross entanglement: connect opposite qubits
for i in range(4):
    ANSATZ_SPEC.append({"gate": "CNOT", "wires": [i, i + 4]})

# Layer 3: Final RZ rotation
for i in range(8):
    ANSATZ_SPEC.append({"gate": "RZ", "wire": i, "param": f"rz2_{i}"})
