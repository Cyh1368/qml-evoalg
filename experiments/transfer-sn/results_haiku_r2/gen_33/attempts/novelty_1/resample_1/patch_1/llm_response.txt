2. **Same resource usage**: Eight CZ gates per block maintained, so depth and gate_count remain constant.

def apply_ansatz_block(params, block_index: int) -> None:
    """Apply one candidate ansatz block from the formal ANSATZ_SPEC.
    
    For CZ gates: alternate between two ring patterns based on block_index parity.
    Even blocks: standard ring [0,1], [1,2], ..., [7,0]
    Odd blocks: shifted ring [1,2], [2,3], ..., [0,7] (incremented by 1 mod 8)
    """
    if SPEC_ERRORS:
        raise ValueError("; ".join(SPEC_ERRORS))

    offset = block_index * N_PARAMS_PER_BLOCK
    key_to_position = {key: i for i, key in enumerate(PARAMETER_KEYS_PER_BLOCK)}
    
    # Determine CZ shift based on block parity for dynamic rewiring
    cz_shift = 0 if block_index % 2 == 0 else 1
    
    for item in ANSATZ_SPEC:
        gate = str(item["gate"]).upper()
        if gate in ALLOWED_SINGLE_QUBIT_GATES:
            angle = params[offset + key_to_position[item["param"]]]
            wire = int(item["wire"])
            if gate == "RX":
                qml.RX(angle, wires=wire)
            elif gate == "RY":
                qml.RY(angle, wires=wire)
            elif gate == "RZ":
                qml.RZ(angle, wires=wire)
        elif gate in ALLOWED_TWO_QUBIT_GATES:
            if gate == "CNOT":
                wires = [int(w) for w in item["wires"]]
                qml.CNOT(wires=wires)
            elif gate == "CZ":
                # Apply dynamic rewiring for CZ gates
                base_wires = [int(w) for w in item["wires"]]
                # Shift both wires by cz_shift, wrapping around N_QUBITS
                shifted_wires = [(w + cz_shift) % N_QUBITS for w in base_wires]
                qml.CZ(wires=shifted_wires)
        elif gate in ALLOWED_PARAM_TWO_QUBIT_GATES:
            angle = params[offset + key_to_position[item["param"]]]
            wires = [int(w) for w in item["wires"]]
            if gate == "CRX":
                qml.CRX(angle, wires=wires)
            elif gate == "CRY":
                qml.CRY(angle, wires=wires)
            elif gate == "CRZ":
                qml.CRZ(angle, wires=wires)
