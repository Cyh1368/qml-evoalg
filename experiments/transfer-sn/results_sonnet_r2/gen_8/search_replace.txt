def _build_ansatz_spec() -> list[dict]:
    """Build an ansatz block: RY+RZ single-qubit layer, ring+skip CRZ
    entangling layer with trainable strengths, and a final RY layer.

    Rationale: the seed used a fixed CZ line (no trainable entanglement
    strength) plus a redundant second RZ layer. Here we replace the fixed
    CZ chain with trainable CRZ gates arranged in a ring (covers all
    qubits with wraparound) plus a couple of long-range skip connections
    to increase effective connectivity without adding many parameters.
    The final single-qubit layer uses RY (a different rotation axis than
    the Z-heavy feature map and entangler) to maximize expressivity per
    parameter, instead of a second RZ which is partially redundant given
    the IsingZZ feature map and CRZ entanglers both act around Z.
    """
    spec: list[dict] = []

    # Pre-entanglement single-qubit layer: RY then RZ per qubit.
    for w in range(N_QUBITS):
        spec.append({"gate": "RY", "wire": w, "param": f"ry_pre_{w}"})
    for w in range(N_QUBITS):
        spec.append({"gate": "RZ", "wire": w, "param": f"rz_pre_{w}"})

    # Ring of trainable CRZ entanglers (covers all qubits with wraparound).
    for w in range(N_QUBITS):
        target = (w + 1) % N_QUBITS
        spec.append({"gate": "CRZ", "wires": [w, target], "param": f"crz_ring_{w}"})

    # A few long-range skip connections for richer connectivity.
    skip_pairs = [(0, 4), (2, 6)]
    for idx, (a, b) in enumerate(skip_pairs):
        spec.append({"gate": "CRZ", "wires": [a, b], "param": f"crz_skip_{idx}"})

    # Post-entanglement single-qubit layer: RY (different axis than RZ/CRZ).
    for w in range(N_QUBITS):
        spec.append({"gate": "RY", "wire": w, "param": f"ry_post_{w}"})

    return spec

def _build_ansatz_spec() -> list[dict]:
    """Build a lean ansatz block combining the fast-converging seed's
    single-qubit layout (RY+RZ pre, RZ post) with a fixed CZ ring
    (wraparound closure added) plus two shared-parameter long-range CRZ
    skip connectors.

    Rationale: the best-scoring prior candidate used RY+RZ pre-rotation,
    a fixed (non-trainable) CZ chain, and a final RZ layer -- only 146
    total params, converging in 750 steps. The fully-trainable CRZ
    ring+skip variant added too many independent parameters (206) and
    lost the fast, stable convergence. Here we keep the cheap fixed CZ
    entangling backbone (which acts like a structural prior well matched
    to the IsingZZ feature map) but close the ring with a wraparound CZ
    (7->0) for full-cycle connectivity, and add exactly one extra
    trainable parameter shared across two long-range CRZ skip gates
    (0-4 and 2-6) to inject a bit of extra long-range, tunable coupling
    without materially increasing the parameter budget or slowing
    convergence.
    """
    spec: list[dict] = []

    # Pre-entanglement single-qubit layer: RY then RZ per qubit.
    for w in range(N_QUBITS):
        spec.append({"gate": "RY", "wire": w, "param": f"ry_pre_{w}"})
    for w in range(N_QUBITS):
        spec.append({"gate": "RZ", "wire": w, "param": f"rz_pre_{w}"})

    # Fixed CZ ring (nearest-neighbor chain plus wraparound closure).
    for w in range(N_QUBITS):
        target = (w + 1) % N_QUBITS
        spec.append({"gate": "CZ", "wires": [w, target]})

    # Two long-range skip connectors sharing a single trainable parameter.
    skip_pairs = [(0, 4), (2, 6)]
    for a, b in skip_pairs:
        spec.append({"gate": "CRZ", "wires": [a, b], "param": "crz_skip_shared"})

    # Post-entanglement single-qubit layer: RZ (matches feature map axis).
    for w in range(N_QUBITS):
        spec.append({"gate": "RZ", "wire": w, "param": f"rz_post_{w}"})

    return spec
