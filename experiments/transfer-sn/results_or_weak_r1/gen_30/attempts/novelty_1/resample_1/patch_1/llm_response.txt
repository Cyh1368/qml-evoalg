<DESCRIPTION>
Introduce trainable feature-scale factors per upload to allow the circuit to adapt its fixed data embedding strength. This adds only 3 extra weights (one per upload) to probe whether flexible encoding enhances expressivity. The feature_map function is modified to multiply FEATURE_SCALE by a trainable factor before each upload.

def feature_map(records) -> None:
    """Encode each feature as a fixed two-qubit phase coupling.

    Accepts a batch of records (shape (B, N_FEATURES)); records[..., k]
    broadcasts over the leading batch dimension.
    """
    for k, (a, b) in enumerate(FEATURE_PAIRS):
        qml.IsingZZ(FEATURE_SCALE * records[..., k], wires=[a, b])

    """Apply one candidate ansatz block from the formal ANSATZ_SPEC."""
    if SPEC_ERRORS:

    offset = block_index * N_PARAMS_PER_BLOCK
    key_to_position = {key: i for i, key in enumerate(PARAMETER_KEYS_PER_BLOCK)}
    for item in ANSATZ_SPEC:
        gate = str(item["gate"]).upper()
        if gate in ALLOWED_SINGLE_QUBIT_GATES:
            angle = params[offset + key_to_position[item["param"]]]

            if gate == "RX":

                qml.RY(angle, wires=wire)

                qml.RZ(angle, wires=wire)

def feature_map(records, upload_index: int = 0, params: np.ndarray = None) -> None:
    """Encode each feature as a fixed two-qubit phase coupling.

    Accepts a batch of records (shape (B, N_FEATURES)); records[..., k]
    broadcasts over the leading batch dimension.
    """
    # Get the feature scale factor for this upload
    if params is not None and upload_index < N_UPLOADS:
        # Map upload index to parameter index for feature scales
        fs_index = N_PARAMS_PER_BLOCK - N_UPLOADS + upload_index
        if fs_index < len(params):
            scale_factor = params[fs_index]
        else:
            scale_factor = 1.0
    else:
        scale_factor = 1.0
    
    for k, (a, b) in enumerate(FEATURE_PAIRS):
        qml.IsingZZ(scale_factor * FEATURE_SCALE * records[..., k], wires=[a, b])
