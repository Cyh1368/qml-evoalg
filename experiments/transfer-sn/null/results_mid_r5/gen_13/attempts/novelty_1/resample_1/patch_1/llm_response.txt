<DESCRIPTION>
Upgrade the global readout calibration (a single gain and bias) to a per-qubit gain and bias system. By applying a unique trainable linear transformation to each qubit's Z-expectation before averaging, the model can better account for the varied roles qubits play in the multi-upload circuit. This increases the total number of readout parameters from 2 to 16 (for 8 qubits), which is a small increase compared to the overall parameter count but significantly improves the model's ability to minimize the loss and close the generalization gap.
</DESCRIPTION>

def predict_batch(params, x_batch: np.ndarray):
    """Scalar prediction per record: calibrated mean of qubit Z expectations."""
    circuit_params = params[:N_CIRCUIT_PARAMS]
    gains = params[N_CIRCUIT_PARAMS : N_CIRCUIT_PARAMS + N_QUBITS]
    biases = params[N_CIRCUIT_PARAMS + N_QUBITS : N_CIRCUIT_PARAMS + 2 * N_QUBITS]
    records = pnp.array(np.asarray(x_batch, dtype=float), requires_grad=False)
    z = pnp.stack(circuit_z_expectations(records, circuit_params))  # (N_QUBITS, B)
    # Apply per-qubit calibration before averaging across qubits
    calibrated = gains[:, None] * (z - biases[:, None])  # broadcasting: (N_Q, 1) * ((N_Q, B) - (N_Q, 1))
    return pnp.mean(calibrated, axis=0)
