<DESCRIPTION>
The ansatz architecture was redesigned to improve information propagation and parameter efficiency. The new structure uses:
1. **Full-Width RY Layer**: Eight independent RY rotations to provide initial state flexibility.
2. **Circular CNOT Chain**: A complete 0-7, 7-0 chain for robust local entanglement.
3. **Multi-Scale Controlled Rotations**: A two-tier long-range interaction strategy using CRY gates for distance-4 pairs (e.g., 0-4) and CRZ gates for distance-2 pairs (e.g., 0-2). This captures both local and global correlations from the 28-feature input.
4. **Selective Parameter Sharing**: A final RZ layer that shares a single parameter across all qubits, significantly reducing the total parameter count from 146 to 104.
This structural approach aims to decrease the generalization gap and improve parameter efficiency while maintaining high classification accuracy.

```python
import json
import os
from pathlib import Path
import numpy as np
import pennylane as qml
from pennylane import numpy as pnp

# Problem constants
N_QUBITS = 8
N_UPLOADS = 3
N_REPEATS = 2
FEATURE_SCALE = np.pi / 2

_DATA_PATH = os.environ.get("TASK_DATA", str(Path(__file__).resolve().parent / "dataset.npz"))
_DATA = np.load(_DATA_PATH)
FEATURE_PAIRS = [tuple(int(w) for w in row) for row in _DATA["feature_pairs"]]
N_FEATURES = len(FEATURE_PAIRS)

ALLOWED_SINGLE_QUBIT_GATES = {"RX", "RY", "RZ"}
ALLOWED_TWO_QUBIT_GATES = {"CNOT", "CZ"}
ALLOWED_PARAM_TWO_QUBIT_GATES = {"CRX", "CRY", "CRZ"}

# EVOLVE-BLOCK-START
ANSATZ_SPEC = [
    # Layer 1: Independent RY rotations for initial expressiveness
    {"gate": "RY", "wire": 0, "param": "ry0"},
    {"gate": "RY", "wire": 1, "param": "ry1"},
    {"gate": "RY", "wire": 2, "param": "ry2"},
    {"gate": "RY", "wire": 3, "param": "ry3"},
    {"gate": "RY", "wire": 4, "param": "ry4"},
    {"gate": "RY", "wire": 5, "param": "ry5"},
    {"gate": "RY", "wire": 6, "param": "ry6"},
    {"gate": "RY", "wire": 7, "param": "ry7"},
    
    # Layer 2: Circular CNOT chain for immediate information flow
    {"gate": "CNOT", "wires": [0, 1]},
    {"gate": "CNOT", "wires": [1, 2]},
    {"gate": "CNOT", "wires": [2, 3]},
    {"gate": "CNOT", "wires": [3, 4]},
    {"gate": "CNOT", "wires": [4, 5]},
    {"gate": "CNOT", "wires": [5, 6]},
    {"gate": "CNOT", "wires": [6, 7]},
    {"gate": "CNOT", "wires": [7, 0]},
    
    # Layer 3: Long-range CRY interactions (distance 4)
    {"gate": "CRY", "wires": [0, 4], "param": "cry0"},
    {"gate": "CRY", "wires": [1, 5], "param": "cry1"},
    {"gate": "CRY", "wires": [2, 6], "param": "cry2"},
    {"gate": "CRY", "wires": [3, 7], "param": "cry3"},
    
    # Layer 4: Medium-range CRZ interactions (distance 2)
    {"gate": "CRZ", "wires": [0, 2], "param": "crz0"},
    {"gate": "CRZ", "wires": [1, 3], "param": "crz1"},
    {"gate": "CRZ", "wires": [4, 6], "param": "crz2"},
    {"gate": "CRZ", "wires": [5, 7], "param": "crz3"},
    
    # Layer 5: Shared RZ layer to refine phases globally
    {"gate": "RZ", "wire": 0, "param": "rz_all"},
    {"gate": "RZ", "wire": 1, "param": "rz_all"},
    {"gate": "RZ", "wire": 2, "param": "rz_all"},
    {"gate": "RZ", "wire": 3, "param": "rz_all"},
    {"gate": "RZ", "wire": 4, "param": "rz_all"},
    {"gate": "RZ", "wire": 5, "param": "rz_all"},
    {"gate": "RZ", "wire": 6, "param": "rz_all"},
    {"gate": "RZ", "wire": 7, "param": "rz_all"},
]
# EVOLVE-BLOCK-END

def load_splits() -> dict:
    return {
        "train": (np.asarray(_DATA["x_train"], dtype=float), np.asarray(_DATA["y_train"], dtype=float)),
        "validation": (np.asarray(_DATA["x_validation"], dtype=float), np.asarray(_DATA["y_validation"], dtype=float)),
        "test": (np.asarray(_DATA["x_test"], dtype=float), np.asarray(_DATA["y_test"], dtype=float)),
    }

def validate_ansatz_spec(spec: list[dict]) -> tuple[list[str], list[str]]:
    errors = []
    parameter_keys = []
    if not isinstance(spec, list) or not spec:
        return ["ANSATZ_SPEC must be a non-empty list of gate dictionaries."], []
    for gate_index, item in enumerate(spec):
        prefix = f"gate {gate_index}"
        if not isinstance(item, dict):
            errors.append(f"{prefix}: expected a dict, got {type(item).__name__}")
            continue
        gate = str(item.get("gate", "")).upper()
        if gate in ALLOWED_SINGLE_QUBIT_GATES:
            wire = item.get("wire")
            if not isinstance(wire, int) or not 0 <= wire < N_QUBITS:
                errors.append(f"{prefix}: {gate} requires integer wire in [0, {N_QUBITS - 1}]")
            param = item.get("param")
            if not isinstance(param, str) or not param:
                errors.append(f"{prefix}: {gate} requires a non-empty string param key")
            else:
                parameter_keys.append(param)
        elif gate in ALLOWED_TWO_QUBIT_GATES | ALLOWED_PARAM_TWO_QUBIT_GATES:
            wires = item.get("wires")
            if (not isinstance(wires, (list, tuple)) or len(wires) != 2 
                or not all(isinstance(w, int) for w in wires) or wires[0] == wires[1]
                or not all(0 <= w < N_QUBITS for w in wires)):
                errors.append(f"{prefix}: {gate} requires two distinct integer wires")
            if gate in ALLOWED_PARAM_TWO_QUBIT_GATES:
                param = item.get("param")
                if not isinstance(param, str) or not param:
                    errors.append(f"{prefix}: {gate} requires a non-empty string param key")
                else:
                    parameter_keys.append(param)
        else:
            allowed = sorted(ALLOWED_SINGLE_QUBIT_GATES | ALLOWED_TWO_QUBIT_GATES | ALLOWED_PARAM_TWO_QUBIT_GATES)
            errors.append(f"{prefix}: unsupported gate {gate!r}; allowed gates are {allowed}")
    return errors, list(dict.fromkeys(parameter_keys))

SPEC_ERRORS, PARAMETER_KEYS_PER_BLOCK = validate_ansatz_spec(ANSATZ_SPEC)
N_PARAMS_PER_BLOCK = len(PARAMETER_KEYS_PER_BLOCK)
N_CIRCUIT_PARAMS = N_PARAMS_PER_BLOCK * N_UPLOADS * N_REPEATS
N_READOUT_PARAMS = 2
N_PARAMS = N_CIRCUIT_PARAMS + N_READOUT_PARAMS

def feature_map(records) -> None:
    for k, (a, b) in enumerate(FEATURE_PAIRS):
        qml.IsingZZ(FEATURE_SCALE * records[..., k], wires=[a, b])

def apply_ansatz_block(params, block_index: int) -> None:
    if SPEC_ERRORS: raise ValueError("; ".join(SPEC_ERRORS))
    offset = block_index * N_PARAMS_PER_BLOCK
    key_to_pos = {key: i for i, key in enumerate(PARAMETER_KEYS_PER_BLOCK)}
    for item in ANSATZ_SPEC:
        gate = str(item["gate"]).upper()
        if gate in ALLOWED_SINGLE_QUBIT_GATES:
            angle = params[offset + key_to_pos[item["param"]]]
            wire = int(item["wire"])
            if gate == "RX": qml.RX(angle, wires=wire)
            elif gate == "RY": qml.RY(angle, wires=wire)
            elif gate == "RZ": qml.RZ(angle, wires=wire)
        elif gate in ALLOWED_TWO_QUBIT_GATES:
            wires = [int(w) for w in item["wires"]]
            if gate == "CNOT": qml.CNOT(wires=wires)
            elif gate == "CZ": qml.CZ(wires=wires)
        elif gate in ALLOWED_PARAM_TWO_QUBIT_GATES:
            angle = params[offset + key_to_pos[item["param"]]]
            wires = [int(w) for w in item["wires"]]
            if gate == "CRX": qml.CRX(angle, wires=wires)
            elif gate == "CRY": qml.CRY(angle, wires=wires)
            elif gate == "CRZ": qml.CRZ(angle, wires=wires)

_DEVICE = qml.device("default.qubit", wires=N_QUBITS, shots=None)

@qml.qnode(_DEVICE, interface="autograd", diff_method="backprop")
def circuit_z_expectations(records, circuit_params):
    for upload_index in range(N_UPLOADS):
        feature_map(records)
        for repeat_index in range(N_REPEATS):
            block_index = upload_index * N_REPEATS + repeat_index
            apply_ansatz_block(circuit_params, block_index)
    return [qml.expval(qml.PauliZ(wire)) for wire in range(N_QUBITS)]

def predict_batch(params, x_batch: np.ndarray):
    circuit_params = params[:N_CIRCUIT_PARAMS]
    gain, bias = params[N_CIRCUIT_PARAMS], params[N_CIRCUIT_PARAMS + 1]
    records = pnp.array(np.asarray(x_batch, dtype=float), requires_grad=False)
    z = pnp.stack(circuit_z_expectations(records, circuit_params))
    raw = pnp.mean(z, axis=0)
    return gain * (raw - bias)

def l2_loss(params, x_batch, y_batch):
    predictions = predict_batch(params, x_batch)
    targets = pnp.array(y_batch, requires_grad=False)
    return pnp.mean((predictions - targets) ** 2)

def evaluate_split(params, x, y):
    predictions = np.asarray(predict_batch(params, x), dtype=float)
    loss = float(np.mean((predictions - y) ** 2))
    accuracy = float(np.mean(np.sign(predictions) == np.sign(y)))
    return {"accuracy": accuracy, "loss": loss}

def _batch_indices(order, step, batch_size):
    start = (step * batch_size) % len(order)
    end = start + batch_size
    return order[start:end] if end <= len(order) else np.concatenate([order[start:], order[:end - len(order)]])

def _log(record, log_file, verbose):
    line = json.dumps(record, sort_keys=True)
    if verbose: print(line)
    if log_file:
        with log_file.open("a", encoding="utf-8") as h: h.write(line + "\n")

def circuit_metadata():
    with qml.queuing.AnnotatedQueue() as queue:
        for u in range(N_UPLOADS):
            feature_map(np.zeros((1, N_FEATURES)))
            for r in range(N_REPEATS):
                apply_ansatz_block(np.zeros(max(N_CIRCUIT_PARAMS, 1)), u * N_REPEATS + r)
    tape = qml.tape.QuantumScript.from_queue(queue)
    ops = [(op.name, op.wires.tolist()) for op in tape.operations]
    depth = tape.graph.get_depth() if hasattr(tape.graph, "get_depth") else len(ops)
    return {"operations": ops, "depth": int(depth), "gate_count": int(len(ops))}

def run_experiment(seed=0, n_epochs=None, steps_per_epoch=None, batch_size=None, learning_rate=None, **kwargs):
    if SPEC_ERRORS: return {"spec_valid": False, "error": "; ".join(SPEC_ERRORS)}
    n_epochs = int(os.environ.get("N_EPOCHS", "1000"))
    steps_per_epoch = int(os.environ.get("STEPS_PER_EPOCH", "30"))
    batch_size = int(os.environ.get("BATCH_SIZE", "15"))
    learning_rate = float(os.environ.get("LEARNING_RATE", "0.03"))
    conv_threshold = float(os.environ.get("CONVERGENCE_THRESHOLD", "0.90"))
    early_stopping, patience, min_delta = True, 75, 1e-4

    splits = load_splits()
    x_train, y_train = splits["train"]
    x_val, y_val = splits["validation"]
    x_test, y_test = splits["test"]

    rng = np.random.default_rng(seed)
    init = rng.uniform(-0.05, 0.05, size=N_PARAMS)
    init[N_CIRCUIT_PARAMS], init[N_CIRCUIT_PARAMS + 1] = 1.0, 0.0
    params = pnp.array(init, requires_grad=True)
    optimizer = qml.AdamOptimizer(stepsize=learning_rate)
    
    best_val_loss, best_params, epochs_no_improve = float("inf"), params.copy(), 0
    history = []
    
    for epoch in range(1, n_epochs + 1):
        order = rng.permutation(len(x_train))
        for step in range(steps_per_epoch):
            ids = _batch_indices(order, step, batch_size)
            params, _ = optimizer.step_and_cost(lambda p: l2_loss(p, x_train[ids], y_train[ids]), params)
        
        val_m = evaluate_split(params, x_val, y_val)
        history.append({"epoch": epoch, "validation_accuracy": val_m["accuracy"], "validation_loss": val_m["loss"]})
        
        if val_m["loss"] < best_val_loss - min_delta:
            best_val_loss, best_params, epochs_no_improve = val_m["loss"], params.copy(), 0
        else:
            epochs_no_improve += 1
            if epochs_no_improve >= patience: break

    params = best_params
    f_train, f_val, f_test = evaluate_split(params, x_train, y_train), evaluate_split(params, x_val, y_val), evaluate_split(params, x_test, y_test)
    meta = circuit_metadata()
    
    return {
        "spec_valid": True, "n_qubits": N_QUBITS, "n_params": N_PARAMS,
        "depth": meta["depth"], "gate_count": meta["gate_count"],
        "train_accuracy": f_train["accuracy"], "validation_accuracy": f_val["accuracy"], "test_accuracy": f_test["accuracy"],
        "train_loss": f_train["loss"], "validation_loss": f_val["loss"], "test_loss": f_test["loss"],
        "generalization_gap": abs(f_train["accuracy"] - f_test["accuracy"]),
        "parameter_efficiency": f_test["accuracy"] / N_PARAMS,
        "history_tail": history[-10:]
    }
```
