This should improve mixing/expressivity for global correlations without meaningfully increasing parameter count or depth.

def _diameter_crz(n, even_param="crz_diam_even", odd_param="crz_diam_odd"):
    """Antipodal (distance n/2) entanglement with alternating shared
    parameters. Splitting the single shared diameter knob into an
    even/odd pair (mirroring the ring entanglement's alternating-strength
    motif) gives the global antipodal connections a bit more expressive
    freedom while still costing only two parameters total."""
    half = n // 2
    gates = []
    for i in range(half):
        param = even_param if i % 2 == 0 else odd_param
        gates.append({"gate": "CRZ", "wires": [i, i + half], "param": param})
    return gates
