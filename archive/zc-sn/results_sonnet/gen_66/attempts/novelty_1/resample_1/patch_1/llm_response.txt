    gates = []

        j = (i + 1) % n

def _ring_crz(n, even_param="crz_ring_even", odd_param="cry_ring_odd"):
    """Nearest-neighbor ring entanglement with alternating shared strength
    and alternating rotation axis: even edges use CRZ, odd edges use CRY.
    This axis diversity boosts expressivity of the local entangling layer
    without adding any gates or parameters beyond the original two-param
    scheme."""
    gates = []
    for i in range(n):
        j = (i + 1) % n
        if i % 2 == 0:
            gates.append({"gate": "CRZ", "wires": [i, j], "param": even_param})
        else:
            gates.append({"gate": "CRY", "wires": [i, j], "param": odd_param})
    return gates
