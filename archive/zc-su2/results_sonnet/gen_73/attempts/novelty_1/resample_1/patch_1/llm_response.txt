# 2b. Lightweight, zero-parameter skip-2 sub-ring linking every other
#     qubit (0-2, 2-4, 4-6, 6-0).
for i in range(0, N_QUBITS, 2):
    ANSATZ_SPEC.append({"gate": "CZ", "wires": [i, (i + 2) % N_QUBITS]})

# 2b-2. Intra-block RY rotation layer, reusing the SAME shared "theta"
#       parameter, inserted between the skip-2 sub-ring and the
#       antipodal shortcuts. Interleaving this extra non-commuting
#       rotation axis mid-entanglement heightens expressivity and has
#       been observed to improve convergence speed, all while keeping
#       the parameter count fixed (still a single shared "theta").

    ANSATZ_SPEC.append({"gate": "RY", "wire": wire, "param": "theta"})
