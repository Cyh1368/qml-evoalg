ANSATZ_SPEC = []

# 1. Pre-entanglement rotation layers: ALL 8 qubits share a SINGLE
#    trainable parameter "theta" across two non-commuting axes (RX then
#    RY), maximizing single-qubit expressivity while keeping the total
#    unique-parameter count at exactly 1.
for wire in range(N_QUBITS):
    ANSATZ_SPEC.append({"gate": "RX", "wire": wire, "param": "theta"})
for wire in range(N_QUBITS):
    ANSATZ_SPEC.append({"gate": "RY", "wire": wire, "param": "theta"})

# 2. Fixed nearest-neighbor entangling ring alternating CNOT and CZ gates
#    (zero trainable parameters). Alternating gate types perturbs the
#    entanglement pattern relative to a uniform CZ or CNOT ring, per
#    dense-graph connectivity principles, without adding any parameters.
for i in range(N_QUBITS):
    gate = "CNOT" if i % 2 == 0 else "CZ"
