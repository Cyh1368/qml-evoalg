"""Parametrized quantum ansatz with shared local mixing and tunable
long-range entanglement.
"""

from _backend import run_experiment as _run

N_QUBITS = 8
ALLOWED_SINGLE_QUBIT_GATES = {"RX", "RY", "RZ"}
ALLOWED_TWO_QUBIT_GATES = {"CNOT", "CZ"}
ALLOWED_PARAM_TWO_QUBIT_GATES = {"CRX", "CRY", "CRZ"}
ALLOWED_ISING_GATES = {"XX", "YY", "ZZ"}


# EVOLVE-BLOCK-START
ANSATZ_SPEC = [
    # One shared mixer parameter preserves parameter economy while alternating
    # axes prevents all wires from undergoing an identical transformation.
    {"gate": "RY", "wire": 0, "param": "shared_mix"},
    {"gate": "RX", "wire": 1, "param": "shared_mix"},
    {"gate": "RY", "wire": 2, "param": "shared_mix"},
    {"gate": "RX", "wire": 3, "param": "shared_mix"},
    {"gate": "RY", "wire": 4, "param": "shared_mix"},
    {"gate": "RX", "wire": 5, "param": "shared_mix"},
    {"gate": "RY", "wire": 6, "param": "shared_mix"},
    {"gate": "RX", "wire": 7, "param": "shared_mix"},

    # Brickwork CZ layers propagate information around the ring at low depth.
    {"gate": "CZ", "wires": [0, 1]},
    {"gate": "CZ", "wires": [2, 3]},
    {"gate": "CZ", "wires": [4, 5]},
    {"gate": "CZ", "wires": [6, 7]},

    {"gate": "CZ", "wires": [1, 2]},
    {"gate": "CZ", "wires": [3, 4]},
    {"gate": "CZ", "wires": [5, 6]},
    {"gate": "CZ", "wires": [7, 0]},

    # A single shared parameter controls all long-range correlations.
    {"gate": "CRZ", "wires": [0, 4], "param": "crz_long"},
    {"gate": "CRZ", "wires": [1, 5], "param": "crz_long"},
    {"gate": "CRZ", "wires": [2, 6], "param": "crz_long"},
    {"gate": "CRZ", "wires": [3, 7], "param": "crz_long"},
]
# EVOLVE-BLOCK-END


def run_experiment(**kwargs):
    return _run(ANSATZ_SPEC, **kwargs)
