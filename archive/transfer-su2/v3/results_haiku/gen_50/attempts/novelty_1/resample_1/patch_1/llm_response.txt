def apply_ansatz_block(params, block_index: int) -> None:
    """Apply one candidate ansatz block from the formal ANSATZ_SPEC.

    Parameters are shared across repetitions. Two-qubit gates are circularly
    shifted based on block_index to break symmetry and improve generalization.
    """
    if SPEC_ERRORS:
        raise ValueError("; ".join(SPEC_ERRORS))

    key_to_position = {key: i for i, key in enumerate(PARAMETER_KEYS_PER_BLOCK)}
    shift = block_index % N_QUBITS  # circular shift amount for this repeat
    
    for item in ANSATZ_SPEC:
        gate = str(item["gate"]).upper()
        if gate in ALLOWED_SINGLE_QUBIT_GATES:
            angle = params[key_to_position[item["param"]]]
            wire = int(item["wire"])
            if gate == "RX":
                qml.RX(angle, wires=wire)
            elif gate == "RY":
                qml.RY(angle, wires=wire)
            elif gate == "RZ":
                qml.RZ(angle, wires=wire)
        elif gate in ALLOWED_TWO_QUBIT_GATES:
            wires = [int(w) for w in item["wires"]]
            # Apply shift to fixed two-qubit gates as well
            shifted_wires = [(w + shift) % N_QUBITS for w in wires]
            if gate == "CNOT":
                qml.CNOT(wires=shifted_wires)
            elif gate == "CZ":
                qml.CZ(wires=shifted_wires)
        elif gate in ALLOWED_PARAM_TWO_QUBIT_GATES:
            angle = params[key_to_position[item["param"]]]
            wires = [int(w) for w in item["wires"]]
            # Circularly shift the wires based on block_index
            shifted_wires = [(w + shift) % N_QUBITS for w in wires]
            if gate == "CRX":
                qml.CRX(angle, wires=shifted_wires)
            elif gate == "CRY":
                qml.CRY(angle, wires=shifted_wires)
            elif gate == "CRZ":
                qml.CRZ(angle, wires=shifted_wires)
            elif gate == "XX":
                qml.IsingXX(angle, wires=shifted_wires)
            elif gate == "YY":
                qml.IsingYY(angle, wires=shifted_wires)
            elif gate == "ZZ":
                qml.IsingZZ(angle, wires=shifted_wires)
