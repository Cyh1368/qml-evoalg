def apply_ansatz_block(params, block_index: int) -> None:
    """Apply one candidate ansatz block from the formal ANSATZ_SPEC.

    Parameters are shared across repetitions: every application of the block
    reads the same values. Wire connectivity is cyclically rotated per repeat
    to improve robustness and reduce seed-to-seed variance while maintaining
    parameter economy.
    """
    if SPEC_ERRORS:
        raise ValueError("; ".join(SPEC_ERRORS))

    key_to_position = {key: i for i, key in enumerate(PARAMETER_KEYS_PER_BLOCK)}
    # Cyclic rotation: shift qubit indices by block_index to vary entangling
    # pattern across the three repetitions without adding parameters.
    shift = block_index % N_QUBITS
    
    for item in ANSATZ_SPEC:
        gate = str(item["gate"]).upper()
        if gate in ALLOWED_SINGLE_QUBIT_GATES:
            angle = params[key_to_position[item["param"]]]
            wire = int(item["wire"])
            # Single-qubit gates: also rotate for symmetry and full expressivity
            rotated_wire = (wire + shift) % N_QUBITS
            if gate == "RX":
                qml.RX(angle, wires=rotated_wire)
            elif gate == "RY":
                qml.RY(angle, wires=rotated_wire)
            elif gate == "RZ":
                qml.RZ(angle, wires=rotated_wire)
        elif gate in ALLOWED_TWO_QUBIT_GATES:
            wires = [int(w) for w in item["wires"]]
