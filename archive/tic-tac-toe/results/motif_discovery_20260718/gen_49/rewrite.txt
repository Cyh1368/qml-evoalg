def board_winner(board):
    for line in WIN_LINES:
        total = int(sum(int(board[i]) for i in line))
        if total == 3: return "cross"
        if total == -3: return "circle"
    return None

def board_label(board):
    return board_winner(board) or "draw"

def enumerate_valid_boards():
    seen = set()
    def visit(board, player):
        if board in seen: return
        seen.add(board)
        if board_winner(board) is not None or all(v != 0 for v in board): return
        for i, v in enumerate(board):
            if v == 0:
                nb = list(board); nb[i] = player
                visit(tuple(nb), -player)
    visit((0,) * N_QUBITS, 1)
    return np.array(sorted(seen), dtype=np.int8)

def make_balanced_split(grouped_boards, size, rng, replace):
    per_class = size // len(CLASS_NAMES)
    xs, ys, labels = [], [], []
    for label in CLASS_NAMES:
        pool = grouped_boards[label]
        idx = rng.choice(len(pool), size=per_class, replace=(replace or per_class > len(pool)))
        xs.append(pool[idx])
        ys.append(np.tile(LABEL_VECTORS[label], (per_class, 1)))
        labels.extend([label] * per_class)
    x, y, l = np.concatenate(xs), np.concatenate(ys), np.array(labels)
    o = rng.permutation(len(x))
    return x[o], y[o], l[o]

def build_data_splits(seed=2027, train_size=450, validation_size=300, test_size=600, replace=True):
    rng = np.random.default_rng(seed)
    boards = enumerate_valid_boards()
    grouped = {n: [] for n in CLASS_NAMES}
    for b in boards: grouped[board_label(b)].append(b)
    grouped = {k: np.array(v, dtype=np.int8) for k, v in grouped.items()}
    return {
        "train": make_balanced_split(grouped, train_size, rng, replace),
        "validation": make_balanced_split(grouped, validation_size, rng, replace),
        "test": make_balanced_split(grouped, test_size, rng, replace),
    }

def validate_ansatz_spec(spec):
    errors, parameter_keys = [], []
    for i, item in enumerate(spec):
        gate = str(item.get("gate", "")).upper()
        if gate in ALLOWED_SINGLE_QUBIT_GATES:
            parameter_keys.append(item["param"])
        elif gate in ALLOWED_PARAM_TWO_QUBIT_GATES:
            parameter_keys.append(item["param"])
        elif gate in ALLOWED_PARAM_THREE_QUBIT_GATES:
            parameter_keys.append(item["param"])
    return errors, list(dict.fromkeys(parameter_keys))

SPEC_ERRORS, PARAMETER_KEYS_PER_BLOCK = validate_ansatz_spec(ANSATZ_SPEC)
N_PARAMS_PER_BLOCK = len(PARAMETER_KEYS_PER_BLOCK)
N_PARAMS = N_PARAMS_PER_BLOCK * N_UPLOADS * N_REPEATS

def feature_map(board):
    for wire in range(N_QUBITS):
        qml.RX(FEATURE_SCALE * board[..., wire], wires=wire)

def apply_ansatz_block(params, block_index):
    offset = block_index * N_PARAMS_PER_BLOCK
    key_to_pos = {key: i for i, key in enumerate(PARAMETER_KEYS_PER_BLOCK)}
    for item in ANSATZ_SPEC:
        gate = item["gate"].upper()
        if gate in ALLOWED_SINGLE_QUBIT_GATES:
            angle = params[offset + key_to_pos[item["param"]]]
            getattr(qml, gate)(angle, wires=item["wire"])
        elif gate in ALLOWED_TWO_QUBIT_GATES:
            getattr(qml, gate)(wires=item["wires"])
        elif gate in ALLOWED_PARAM_TWO_QUBIT_GATES:
            angle = params[offset + key_to_pos[item["param"]]]
            getattr(qml, gate)(angle, wires=item["wires"])
        elif gate == "ZZZ":
            angle = params[offset + key_to_pos[item["param"]]]
            qml.MultiRZ(angle, wires=item["wires"])
        elif gate == "CCRZ":
            angle = params[offset + key_to_pos[item["param"]]]
            qml.ctrl(qml.RZ, control=item["wires"][:2])(angle, wires=item["wires"][2])

_DEVICE = qml.device("default.qubit", wires=N_QUBITS)

@qml.qnode(_DEVICE, interface="autograd", diff_method="backprop")
def circuit_z_expectations(board, params):
    for u in range(N_UPLOADS):
        feature_map(board)
        for r in range(N_REPEATS):
            apply_ansatz_block(params, u * N_REPEATS + r)
    return [qml.expval(qml.PauliZ(i)) for i in range(N_QUBITS)]

def predict_batch(params, x_batch):
    boards = pnp.array(x_batch, requires_grad=False)
    z = pnp.stack(circuit_z_expectations(boards, params))
    cross = (z[1] + z[3] + z[5] + z[7]) / 4.0
    circle = (z[0] + z[2] + z[4] + z[6]) / 4.0
    draw = z[8]
    return pnp.stack([cross, circle, draw], axis=1)

def l2_loss(params, x, y):
    preds = predict_batch(params, x)
    return pnp.mean(pnp.sum((preds - pnp.array(y, requires_grad=False))**2, axis=1))

def evaluate_split(params, x, y):
    preds = np.asarray(predict_batch(params, x), dtype=float)
    loss = np.mean(np.sum((preds - y)**2, axis=1))
    acc = np.mean(np.argmax(preds, axis=1) == np.argmax(y, axis=1))
    return {"accuracy": float(acc), "loss": float(loss)}

def run_experiment(seed=0, **kwargs):
    rng = np.random.default_rng(seed)
    splits = build_data_splits(seed=int(os.environ.get("DATA_SEED", 2027)))
    x_tr, y_tr, _ = splits["train"]
    x_val, y_val, _ = splits["validation"]
    x_te, y_te, _ = splits["test"]

    params = pnp.array(rng.uniform(-0.05, 0.05, size=N_PARAMS), requires_grad=True)
    opt = qml.AdamOptimizer(stepsize=float(os.environ.get("LEARNING_RATE", 0.03)))
    
    best_val_loss, best_params, best_epoch = float("inf"), params.copy(), 0
    patience, epochs_no_improve = 75, 0

    for epoch in range(1, int(os.environ.get("N_EPOCHS", 100)) + 1):
        order = rng.permutation(len(x_tr))
        for step in range(int(os.environ.get("STEPS_PER_EPOCH", 30))):
            idx = order[(step*15)%len(order) : (step*15)%len(order)+15]
            params = opt.step(lambda p: l2_loss(p, x_tr[idx], y_tr[idx]), params)
        
        val_m = evaluate_split(params, x_val, y_val)
        if val_m["loss"] < best_val_loss - 1e-4:
            best_val_loss, best_params, best_epoch, epochs_no_improve = val_m["loss"], params.copy(), epoch, 0
        else:
            epochs_no_improve += 1
            if epochs_no_improve >= patience: break

    params = best_params
    tr, val, te = evaluate_split(params, x_tr, y_tr), evaluate_split(params, x_val, y_val), evaluate_split(params, x_te, y_te)
    
    return {
        "spec_valid": True, "n_params": N_PARAMS,
        "train_accuracy": tr["accuracy"], "validation_accuracy": val["accuracy"], "test_accuracy": te["accuracy"],
        "train_loss": tr["loss"], "validation_loss": val["loss"], "test_loss": te["loss"],
        "generalization_gap": abs(tr["accuracy"] - te["accuracy"]),
        "parameter_efficiency": te["accuracy"] / max(N_PARAMS, 1),
        "best_epoch": best_epoch, "stop_reason": "early_stopping" if epochs_no_improve >= patience else "max_epochs"
    }
