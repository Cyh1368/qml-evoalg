"""Winning-line multi-axis ansatz for ShinkaEvolve tic-tac-toe QML.

This candidate improves on the previous winning-line motif approach by 
diversifying the interaction axes (ZZZ vs CCRZ) and adding a fixed 
entanglement layer to improve feature propagation.
"""

# Problem constants
N_QUBITS = 9
N_UPLOADS = 3
N_REPEATS = 2
FEATURE_SCALE = 2 * np.pi / 3

# Paper indexing:
#   0 -- 1 -- 2
#   |    |    |
#   7 -- 8 -- 3

#   6 -- 5 -- 4

# User-facing class order.  Observables are mapped to this order below.
CLASS_NAMES = ("cross", "circle", "draw")

ALLOWED_SINGLE_QUBIT_GATES = {"RX", "RY", "RZ"}
ALLOWED_TWO_QUBIT_GATES = {"CNOT", "CZ"}
ALLOWED_PARAM_TWO_QUBIT_GATES = {"CRX", "CRY", "CRZ"}
ALLOWED_PARAM_THREE_QUBIT_GATES = {"ZZZ", "CCRZ"}

def board_winner(board: tuple[int, ...] | np.ndarray) -> str | None:
    for line in WIN_LINES:
        total = int(sum(int(board[i]) for i in line))
        if total == 3: return "cross"
        if total == -3: return "circle"
    return None

def board_label(board: tuple[int, ...] | np.ndarray) -> str:
    return board_winner(board) or "draw"

def enumerate_valid_boards() -> np.ndarray:
    seen: set[tuple[int, ...]] = set()
    def visit(board: tuple[int, ...], player: int) -> None:
        if board in seen: return
        seen.add(board)
        if board_winner(board) is not None or all(value != 0 for value in board): return
        for wire, value in enumerate(board):
            if value == 0:
                next_board = list(board)
                next_board[wire] = player
                visit(tuple(next_board), -player)
    visit((0,) * N_QUBITS, 1)
    return np.array(sorted(seen), dtype=np.int8)

def make_balanced_split(grouped_boards, size, rng, replace):
    per_class = size // len(CLASS_NAMES)
    xs, ys, labels = [], [], []
    for label in CLASS_NAMES:
        pool = grouped_boards[label]
        indices = rng.choice(len(pool), size=per_class, replace=(replace or per_class > len(pool)))
        xs.append(pool[indices])
        ys.append(np.tile(LABEL_VECTORS[label], (per_class, 1)))
        labels.extend([label] * per_class)
    x, y, l = np.concatenate(xs), np.concatenate(ys), np.array(labels)
    order = rng.permutation(len(x))
    return x[order], y[order], l[order]

def build_data_splits(seed=2027, train_size=450, validation_size=300, test_size=600, replace=True):
    rng = np.random.default_rng(seed)
    boards = enumerate_valid_boards()
    grouped = {name: [] for name in CLASS_NAMES}
    for board in boards: grouped[board_label(board)].append(board)
    grouped_arrays = {n: np.array(v, dtype=np.int8) for n, v in grouped.items()}
    return {
        "train": make_balanced_split(grouped_arrays, train_size, rng, replace),
        "validation": make_balanced_split(grouped_arrays, validation_size, rng, replace),
        "test": make_balanced_split(grouped_arrays, test_size, rng, replace),
    }

def validate_ansatz_spec(spec):
    errors, parameter_keys = [], []
    for i, item in enumerate(spec):
        gate = str(item.get("gate", "")).upper()
        if gate in ALLOWED_SINGLE_QUBIT_GATES:
            parameter_keys.append(item["param"])
        elif gate in ALLOWED_PARAM_TWO_QUBIT_GATES or gate in ALLOWED_PARAM_THREE_QUBIT_GATES:
            parameter_keys.append(item["param"])
    return errors, list(dict.fromkeys(parameter_keys))

SPEC_ERRORS, PARAMETER_KEYS_PER_BLOCK = validate_ansatz_spec(ANSATZ_SPEC)
N_PARAMS_PER_BLOCK = len(PARAMETER_KEYS_PER_BLOCK)
N_PARAMS = N_PARAMS_PER_BLOCK * N_UPLOADS * N_REPEATS

def feature_map(board):
    for wire in range(N_QUBITS):
        qml.RX(FEATURE_SCALE * board[..., wire], wires=wire)

def apply_ansatz_block(params, block_index):
    offset = block_index * N_PARAMS_PER_BLOCK
    key_to_pos = {key: i for i, key in enumerate(PARAMETER_KEYS_PER_BLOCK)}
    for item in ANSATZ_SPEC:
        gate = str(item["gate"]).upper()
        if gate in ALLOWED_SINGLE_QUBIT_GATES:
            qml.__dict__[gate](params[offset + key_to_pos[item["param"]]], wires=item["wire"])
        elif gate == "CNOT":
            qml.CNOT(wires=item["wires"])
        elif gate == "CZ":
            qml.CZ(wires=item["wires"])
        elif gate in ALLOWED_PARAM_TWO_QUBIT_GATES:
            qml.__dict__[gate](params[offset + key_to_pos[item["param"]]], wires=item["wires"])
        elif gate == "ZZZ":
            qml.MultiRZ(params[offset + key_to_pos[item["param"]]], wires=item["wires"])
        elif gate == "CCRZ":
            qml.ctrl(qml.RZ, control=item["wires"][:2])(params[offset + key_to_pos[item["param"]]], wires=item["wires"][2])

_DEVICE = qml.device("default.qubit", wires=N_QUBITS)

@qml.qnode(_DEVICE, interface="autograd", diff_method="backprop")
def circuit_z_expectations(board, params):
    for u in range(N_UPLOADS):
        feature_map(board)
        for r in range(N_REPEATS):
            apply_ansatz_block(params, u * N_REPEATS + r)
    return [qml.expval(qml.PauliZ(w)) for w in range(N_QUBITS)]

def predict_batch(params, x_batch):
    boards = pnp.array(np.asarray(x_batch, dtype=float), requires_grad=False)
    z = pnp.stack(circuit_z_expectations(boards, params))
    return pnp.stack([(z[1]+z[3]+z[5]+z[7])/4.0, (z[0]+z[2]+z[4]+z[6])/4.0, z[8]], axis=1)

def l2_loss(params, x, y):
    preds = predict_batch(params, x)
    return pnp.mean(pnp.sum((preds - pnp.array(y, requires_grad=False))**2, axis=1))

def evaluate_split(params, x, y):
    preds = np.asarray(predict_batch(params, x), dtype=float)
    return {"accuracy": float(np.mean(np.argmax(preds, 1) == np.argmax(y, 1))), "loss": float(np.mean(np.sum((preds-y)**2, 1)))}

def run_experiment(seed=0, **kwargs):
    if SPEC_ERRORS: return {"spec_valid": False, "error": "; ".join(SPEC_ERRORS)}
    rng = np.random.default_rng(seed)
    splits = build_data_splits(seed=int(os.environ.get("DATA_SEED", 2027)))
    x_train, y_train, _ = splits["train"]
    x_val, y_val, _ = splits["validation"]
    x_test, y_test, _ = splits["test"]
    
    params = pnp.array(rng.uniform(-0.05, 0.05, size=N_PARAMS), requires_grad=True)
    opt = qml.AdamOptimizer(stepsize=float(os.environ.get("LEARNING_RATE", 0.03)))
    
    best_val_loss, best_params, patience, epochs_no_improve = float("inf"), params.copy(), 75, 0
    for epoch in range(1, int(os.environ.get("N_EPOCHS", 100)) + 1):
        order = rng.permutation(len(x_train))
        for step in range(int(os.environ.get("STEPS_PER_EPOCH", 30))):
            idx = order[(step*15)%len(order) : (step*15)%len(order)+15]
            params = opt.step(lambda p: l2_loss(p, x_train[idx], y_train[idx]), params)
        
        val_m = evaluate_split(params, x_val, y_val)
        if val_m["loss"] < best_val_loss - 1e-4:
            best_val_loss, best_params, epochs_no_improve = val_m["loss"], params.copy(), 0
        else:
            epochs_no_improve += 1
            if epochs_no_improve >= patience: break

    params = best_params
    f_train, f_val, f_test = evaluate_split(params, x_train, y_train), evaluate_split(params, x_val, y_val), evaluate_split(params, x_test, y_test)
    return {
        "spec_valid": True, "n_params": N_PARAMS, "train_accuracy": f_train["accuracy"],
        "validation_accuracy": f_val["accuracy"], "test_accuracy": f_test["accuracy"],
        "train_loss": f_train["loss"], "validation_loss": f_val["loss"], "test_loss": f_test["loss"],
        "generalization_gap": abs(f_train["accuracy"] - f_test["accuracy"]),
        "parameter_efficiency": f_test["accuracy"] / max(float(N_PARAMS), 1.0)
    }
