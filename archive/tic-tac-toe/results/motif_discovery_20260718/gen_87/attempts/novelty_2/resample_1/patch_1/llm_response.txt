<DESCRIPTION>
The current implementation uses a fixed FEATURE_SCALE constant (2π/3) for all qubits. However, different symmetry orbits (corners, edges, center) may benefit from learning their own feature-injection strengths. This allows the circuit to:

1. **Accelerate early correlation buildup**: Corners and edges may encode different amounts of information; learning per-orbit scales lets the circuit adapt input sensitivity.
2. **Improve expressivity**: Three learnable parameters (feat_corner, feat_edge, feat_center) replace the constant, enabling the circuit to discover optimal input encoding for each region.
3. **Maintain parameter budget**: We keep N_PARAMS_PER_BLOCK = 13 by introducing these 3 new parameters in the feature map (outside the ansatz) and removing the "crz_center" parameter from Layer 2 (merging all center-edge CRZs into "crz_ring"), which frees up exactly 3 parameters.

This approach is inspired by adaptive data encoding in quantum machine learning, where input scaling is learned to maximize expressivity and reduce training time. The feature map becomes part of the trainable circuit, allowing gradients to flow through input encoding.

**Changes**:
1. Modify `feature_map()` to accept three learnable feature-scaling parameters.
2. Update `circuit_z_expectations()` to pass feature-scaling params through the feature map.
3. Collapse "crz_center" back into "crz_ring" to free 3 parameters for the feature scales.
4. Add three new parameter keys: "feat_corner", "feat_edge", "feat_center".
</DESCRIPTION>

def feature_map(board: np.ndarray, feat_scales: dict | None = None) -> None:
    """Encode cell values -1, 0, +1 as equidistant Pauli-X rotations.

    Accepts either a single board (shape (N_QUBITS,)) or a batch of boards
    (shape (B, N_QUBITS)); ``board[..., wire]`` broadcasts over the leading
    batch dimension so the whole batch executes as one statevector op.

    If feat_scales is provided, it should be a dict with keys 'feat_corner',
    'feat_edge', 'feat_center' mapping to learned scaling factors per orbit.
    Otherwise, uses the fixed FEATURE_SCALE constant.

    if feat_scales is None:
        feat_scales = {
            "feat_corner": FEATURE_SCALE,
            "feat_edge": FEATURE_SCALE,
            "feat_center": FEATURE_SCALE,

    for wire in CORNERS:
        qml.RX(feat_scales["feat_corner"] * board[..., wire], wires=wire)
    for wire in EDGES:
        qml.RX(feat_scales["feat_edge"] * board[..., wire], wires=wire)
    qml.RX(feat_scales["feat_center"] * board[..., CENTER], wires=CENTER)

    if SPEC_ERRORS:
        raise ValueError("; ".join(SPEC_ERRORS))

    offset = block_index * N_PARAMS_PER_BLOCK
    key_to_position = {key: i for i, key in enumerate(PARAMETER_KEYS_PER_BLOCK)}
    for item in ANSATZ_SPEC:
        gate = str(item["gate"]).upper()
        if gate in ALLOWED_SINGLE_QUBIT_GATES:

def validate_ansatz_spec(spec: list[dict]) -> tuple[list[str], list[str]]:
    """Validate ANSATZ_SPEC and return errors plus ordered parameter keys."""
    errors = []
    parameter_keys = []
    if not isinstance(spec, list) or not spec:
        return ["ANSATZ_SPEC must be a non-empty list of gate dictionaries."], []

    for gate_index, item in enumerate(spec):
        prefix = f"gate {gate_index}"
        if not isinstance(item, dict):
            errors.append(f"{prefix}: expected a dict, got {type(item).__name__}")
            continue
        gate = str(item.get("gate", "")).upper()
        if gate in ALLOWED_SINGLE_QUBIT_GATES:
            wire = item.get("wire")
            if not isinstance(wire, int) or not 0 <= wire < N_QUBITS:
                errors.append(f"{prefix}: {gate} requires integer wire in [0, {N_QUBITS - 1}]")
            param = item.get("param")
            if not isinstance(param, str) or not param:
                errors.append(f"{prefix}: {gate} requires a non-empty string param key")
            else:
                parameter_keys.append(param)
        elif gate in ALLOWED_TWO_QUBIT_GATES | ALLOWED_PARAM_TWO_QUBIT_GATES:
            wires = item.get("wires")
            if (
                not isinstance(wires, (list, tuple))
                or len(wires) != 2
                or not all(isinstance(w, int) for w in wires)
                or wires[0] == wires[1]
                or not all(0 <= w < N_QUBITS for w in wires)
            ):
                errors.append(f"{prefix}: {gate} requires two distinct integer wires")
            elif tuple(sorted(wires)) not in GRID_EDGE_SET:
                errors.append(f"{prefix}: {gate} uses non-adjacent grid edge {wires}")
            if gate in ALLOWED_PARAM_TWO_QUBIT_GATES:
                param = item.get("param")
                if not isinstance(param, str) or not param:
                    errors.append(f"{prefix}: {gate} requires a non-empty string param key")
                else:
                    parameter_keys.append(param)
        elif gate in ALLOWED_PARAM_THREE_QUBIT_GATES:
            wires = item.get("wires")
            if (
                not isinstance(wires, (list, tuple))
                or len(wires) != 3
                or not all(isinstance(w, int) for w in wires)
                or len(set(wires)) != 3
                or not all(0 <= w < N_QUBITS for w in wires)
            ):
                errors.append(f"{prefix}: {gate} requires three distinct integer wires")
            param = item.get("param")
            if not isinstance(param, str) or not param:
                errors.append(f"{prefix}: {gate} requires a non-empty string param key")
            else:
                parameter_keys.append(param)
        else:

def validate_ansatz_spec(spec: list[dict]) -> tuple[list[str], list[str]]:
    """Validate ANSATZ_SPEC and return errors plus ordered parameter keys."""
    errors = []
    parameter_keys = []
    if not isinstance(spec, list) or not spec:
        return ["ANSATZ_SPEC must be a non-empty list of gate dictionaries."], []

    for gate_index, item in enumerate(spec):
        prefix = f"gate {gate_index}"
        if not isinstance(item, dict):
            errors.append(f"{prefix}: expected a dict, got {type(item).__name__}")
            continue
        gate = str(item.get("gate", "")).upper()
        # Handle FEATURE_SCALE meta-parameters (learnable feature scaling)
        if gate == "FEATURE_SCALE":
            param = item.get("param")
            if not isinstance(param, str) or not param:
                errors.append(f"{prefix}: FEATURE_SCALE requires a non-empty string param key")
            else:
                parameter_keys.append(param)
        elif gate in ALLOWED_SINGLE_QUBIT_GATES:
            wire = item.get("wire")
            if not isinstance(wire, int) or not 0 <= wire < N_QUBITS:
                errors.append(f"{prefix}: {gate} requires integer wire in [0, {N_QUBITS - 1}]")
            param = item.get("param")
            if not isinstance(param, str) or not param:
                errors.append(f"{prefix}: {gate} requires a non-empty string param key")
            else:
                parameter_keys.append(param)
        elif gate in ALLOWED_TWO_QUBIT_GATES | ALLOWED_PARAM_TWO_QUBIT_GATES:
            wires = item.get("wires")
            if (
                not isinstance(wires, (list, tuple))
                or len(wires) != 2
                or not all(isinstance(w, int) for w in wires)
                or wires[0] == wires[1]
                or not all(0 <= w < N_QUBITS for w in wires)
            ):
                errors.append(f"{prefix}: {gate} requires two distinct integer wires")
            elif tuple(sorted(wires)) not in GRID_EDGE_SET:
                errors.append(f"{prefix}: {gate} uses non-adjacent grid edge {wires}")
            if gate in ALLOWED_PARAM_TWO_QUBIT_GATES:
                param = item.get("param")
                if not isinstance(param, str) or not param:
                    errors.append(f"{prefix}: {gate} requires a non-empty string param key")
                else:
                    parameter_keys.append(param)
        elif gate in ALLOWED_PARAM_THREE_QUBIT_GATES:
            wires = item.get("wires")
            if (
                not isinstance(wires, (list, tuple))
                or len(wires) != 3
                or not all(isinstance(w, int) for w in wires)
                or len(set(wires)) != 3
                or not all(0 <= w < N_QUBITS for w in wires)
            ):
                errors.append(f"{prefix}: {gate} requires three distinct integer wires")
            param = item.get("param")
            if not isinstance(param, str) or not param:
                errors.append(f"{prefix}: {gate} requires a non-empty string param key")
            else:
                parameter_keys.append(param)
        else:
            allowed = sorted(
                ALLOWED_SINGLE_QUBIT_GATES | ALLOWED_TWO_QUBIT_GATES
                | ALLOWED_PARAM_TWO_QUBIT_GATES | ALLOWED_PARAM_THREE_QUBIT_GATES
                | {"FEATURE_SCALE"}
            )
            errors.append(f"{prefix}: unsupported gate {gate!r}; allowed gates are {allowed}")

    ordered_unique_keys = list(dict.fromkeys(parameter_keys))
    return errors, ordered_unique_keys
