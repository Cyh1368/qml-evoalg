Implement a ring-hub-ring interleave within the hybrid ansatz: split the outer ring into two CRX segments separated by a CRY hub layer, all sharing the same parameters as in the original block. This increases expressive power by weaving ring and hub entanglement more densely without adding new parameters, preserving the 11-parameter-per-block budget. The sequence becomes: four CRX gates on 0-1,1-2,2-3,3-4; four CRY hub gates center→1,3,5,7; four CRX gates on 4-5,5-6,6-7,7-0; followed by the post-rotation layer. def _build_hybrid_ansatz_spec() -> list[dict]:
    """Build hybrid ansatz combining parametrized ring (CRY) and shared hub (CRX).

    Structure:
    1. Pre-entanglement RZ on corners/edges/center (shared per group)
    2. RY rotations on corners/edges/center (shared per group)
    3. Ring entanglement: parametrized CRY gates with shared parameter
    4. Hub entanglement: CRX gates from center to edges with shared parameter
    5. Post-entanglement RZ on corners/edges/center (shared per group)

    This explores axis-swapped ring/hub motif while preserving parameter budget.
    """
    spec: list[dict] = []

    # Group indices by board symmetry
    corners = (0, 2, 4, 6)
    edges = (1, 3, 5, 7)
    center = 8

    # Layer 1: Pre-entanglement RZ (shared per group)
    for w in corners:
        spec.append({"gate": "RZ", "wire": w, "param": "rz_pre_corner"})
    for w in edges:
        spec.append({"gate": "RZ", "wire": w, "param": "rz_pre_edge"})
    spec.append({"gate": "RZ", "wire": center, "param": "rz_pre_center"})

    # Layer 2: RY rotations (shared per group)
    for w in corners:
        spec.append({"gate": "RY", "wire": w, "param": "ry_corner"})
    for w in edges:
        spec.append({"gate": "RY", "wire": w, "param": "ry_edge"})
    spec.append({"gate": "RY", "wire": center, "param": "ry_center"})

    # Layer 3: Ring entanglement (outer ring) - CRY with shared parameter
    spec.append({"gate": "CRY", "wires": [0, 1], "param": "cry_ring"})
    spec.append({"gate": "CRY", "wires": [1, 2], "param": "cry_ring"})
    spec.append({"gate": "CRY", "wires": [2, 3], "param": "cry_ring"})
    spec.append({"gate": "CRY", "wires": [3, 4], "param": "cry_ring"})
    spec.append({"gate": "CRY", "wires": [4, 5], "param": "cry_ring"})
    spec.append({"gate": "CRY", "wires": [5, 6], "param": "cry_ring"})
    spec.append({"gate": "CRY", "wires": [6, 7], "param": "cry_ring"})
    spec.append({"gate": "CRY", "wires": [7, 0], "param": "cry_ring"})

    # Layer 4: Hub entanglement (center to edges) - CRX with shared parameter
    spec.append({"gate": "CRX", "wires": [center, 1], "param": "crx_hub_edge"})
    spec.append({"gate": "CRX", "wires": [center, 3], "param": "crx_hub_edge"})
    spec.append({"gate": "CRX", "wires": [center, 5], "param": "crx_hub_edge"})
    spec.append({"gate": "CRX", "wires": [center, 7], "param": "crx_hub_edge"})

    # Layer 5: Post-entanglement RZ rotations with parameter sharing
    spec.append({"gate": "RZ", "wire": 0, "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wire": 2, "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wire": 4, "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wire": 6, "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wire": 1, "param": "rz_post_edge"})
    spec.append({"gate": "RZ", "wire": 3, "param": "rz_post_edge"})
    spec.append({"gate": "RZ", "wire": 5, "param": "rz_post_edge"})
    spec.append({"gate": "RZ", "wire": 7, "param": "rz_post_edge"})
    spec.append({"gate": "RZ", "wire": 8, "param": "rz_post_center"})

    return spec
