def _build_hierarchical_ansatz_spec() -> list[dict]:
    """Build hierarchical entanglement ansatz with parameter sharing.

    Structural changes (improved efficiency and symmetry):
    1) Pre-entanglement RZ on all qubits (shared by symmetry)
    2) RY rotations on all qubits (shared by symmetry)
    3) Ring entanglement: CRX gates around the outer ring with a single shared parameter
    4) Hub entanglement: center-to-edge gates with two shared parameters
       - hub_crz_h shared by edges 1 and 5 (horizontal group)
       - hub_crz_v shared by edges 3 and 7 (vertical group)
    5) Post-entanglement RZ on all qubits (shared by symmetry)
    """
    spec: list[dict] = []

    # Symmetry groups for 4-fold rotational symmetry of tic-tac-toe board
    # Corners: 0, 2, 4, 6 (all equivalent by rotation)
    # Edges: 1, 3, 5, 7 (all equivalent by rotation)
    # Center: 8 (unique)

    # Layer 1: Pre-entanglement RZ rotations with parameter sharing
    spec.append({"gate": "RZ", "wire": 0, "param": "rz_pre_corner"})
    spec.append({"gate": "RZ", "wire": 2, "param": "rz_pre_corner"})
    spec.append({"gate": "RZ", "wire": 4, "param": "rz_pre_corner"})
    spec.append({"gate": "RZ", "wire": 6, "param": "rz_pre_corner"})
    # Edge wires split into horizontal vs vertical groups
    spec.append({"gate": "RZ", "wire": 1, "param": "rz_pre_edge_h"})
    spec.append({"gate": "RZ", "wire": 3, "param": "rz_pre_edge_v"})
    spec.append({"gate": "RZ", "wire": 5, "param": "rz_pre_edge_h"})
    spec.append({"gate": "RZ", "wire": 7, "param": "rz_pre_edge_v"})
    spec.append({"gate": "RZ", "wire": 8, "param": "rz_pre_center"})

    # Layer 2: RY rotations with parameter sharing
    spec.append({"gate": "RY", "wire": 0, "param": "ry_corner"})
    spec.append({"gate": "RY", "wire": 2, "param": "ry_corner"})
    spec.append({"gate": "RY", "wire": 4, "param": "ry_corner"})
    spec.append({"gate": "RY", "wire": 6, "param": "ry_corner"})
    spec.append({"gate": "RY", "wire": 1, "param": "ry_edge_h"})
    spec.append({"gate": "RY", "wire": 3, "param": "ry_edge_v"})
    spec.append({"gate": "RY", "wire": 5, "param": "ry_edge_h"})
    spec.append({"gate": "RY", "wire": 7, "param": "ry_edge_v"})
    spec.append({"gate": "RY", "wire": 8, "param": "ry_center"})

    # Layer 3: Ring entanglement (outer 8 qubits)
    # Parameterized CRX ring on the perimeter (shared angle)
    spec.append({"gate": "CRX", "wires": [0, 1], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [1, 2], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [2, 3], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [3, 4], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [4, 5], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [5, 6], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [6, 7], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [7, 0], "param": "crx_ring"})

    # Layer 4: Hub entanglement (center to edges) with two shared parameters
    # Horizontal edge group (edges 1 and 5)
    spec.append({"gate": "CRZ", "wires": [CENTER, 1], "param": "hub_crz_h"})
    spec.append({"gate": "CRZ", "wires": [CENTER, 5], "param": "hub_crz_h"})
    # Vertical edge group (edges 3 and 7)
    spec.append({"gate": "CRZ", "wires": [CENTER, 3], "param": "hub_crz_v"})
    spec.append({"gate": "CRZ", "wires": [CENTER, 7], "param": "hub_crz_v"})

    # Layer 5: Post-entanglement RZ rotations with parameter sharing
    spec.append({"gate": "RZ", "wires": [0], "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wires": [2], "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wires": [4], "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wires": [6], "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wires": [1], "param": "rz_post_edge_h"})
    spec.append({"gate": "RZ", "wires": [3], "param": "rz_post_edge_v"})
    spec.append({"gate": "RZ", "wires": [5], "param": "rz_post_edge_h"})
    spec.append({"gate": "RZ", "wires": [7], "param": "rz_post_edge_v"})
    spec.append({"gate": "RZ", "wires": [8], "param": "rz_post_center"})

    return spec


ANSATZ_SPEC = _build_hierarchical_ansatz_spec()
