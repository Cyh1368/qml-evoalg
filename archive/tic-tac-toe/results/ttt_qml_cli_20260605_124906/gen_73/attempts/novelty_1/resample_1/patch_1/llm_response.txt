<DESCRIPTION>
Introduce an alternating entanglement ordering between the ring CRX gates and the hub CRX/CRY gates to diversify temporal entanglement patterns and potentially improve optimization and generalization. The new approach preserves parameter sharing and the overall block structure, but for every block decides whether to apply the ring gates before the hub gates or vice versa based on block_index parity.

def apply_ansatz_block(params, block_index: int) -> None:
    """Apply one candidate ansatz block from the formal ANSATZ_SPEC with alternating
    ring/hub entanglement order to diversify temporal entanglement."""
    if SPEC_ERRORS:
        raise ValueError("; ".join(SPEC_ERRORS))

    offset = block_index * N_PARAMS_PER_BLOCK
    key_to_position = {key: i for i, key in enumerate(PARAMETER_KEYS_PER_BLOCK)}

    # Indices for gate groups to allow alternating ordering while preserving
    # parameter mapping. These rely on the existing ANSATZ_SPEC structure.
    ring_indices = [i for i, it in enumerate(ANSATZ_SPEC) if str(it.get("gate","")).upper() == "CRX" and it.get("param","") == "crx_ring"]
    hub_indices = [i for i, it in enumerate(ANSATZ_SPEC) if str(it.get("gate","")).upper() in {"CRX","CRY"} and it.get("param","") in {"hub_crx","hub_cry"}]
    pre_indices = [i for i, it in enumerate(ANSATZ_SPEC) if str(it.get("gate","")).upper() == "RZ" and it.get("param","").startswith("rz_pre")]
    ry_indices = [i for i, it in enumerate(ANSATZ_SPEC) if str(it.get("gate","")).upper() == "RY" and it.get("param","").startswith("ry_")]
    post_indices = [i for i, it in enumerate(ANSATZ_SPEC) if str(it.get("gate","")).upper() == "RZ" and it.get("param","").startswith("rz_post")]

    # Alternate the order of ring vs hub between uploads/repeats
    if (block_index % 2) == 0:
        order = pre_indices + ry_indices + ring_indices + hub_indices + post_indices
    else:
        order = pre_indices + ry_indices + hub_indices + ring_indices + post_indices

    for idx in order:
        item = ANSATZ_SPEC[idx]
        gate = str(item["gate"]).upper()
        if gate in ALLOWED_SINGLE_QUBIT_GATES:
            angle = params[offset + key_to_position[item["param"]]]
            wire = int(item["wire"])
            if gate == "RX":
                qml.RX(angle, wires=wire)
            elif gate == "RY":
                qml.RY(angle, wires=wire)
            elif gate == "RZ":
                qml.RZ(angle, wires=wire)
        elif gate in ALLOWED_TWO_QUBIT_GATES:
            wires = [int(w) for w in item["wires"]]
            if gate == "CNOT":
                qml.CNOT(wires=wires)
            elif gate == "CZ":
                qml.CZ(wires=wires)
        elif gate in ALLOWED_PARAM_TWO_QUBIT_GATES:
            angle = params[offset + key_to_position[item["param"]]]
            wires = [int(w) for w in item["wires"]]
            if gate == "CRX":
                qml.CRX(angle, wires=wires)
            elif gate == "CRY":
                qml.CRY(angle, wires=wires)
            elif gate == "CRZ":
                qml.CRZ(angle, wires=wires)
