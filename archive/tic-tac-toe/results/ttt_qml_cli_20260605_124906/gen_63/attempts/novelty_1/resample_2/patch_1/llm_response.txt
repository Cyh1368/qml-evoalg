<DESCRIPTION>Unify pre- and post-entanglement RZ parameters across corner, edge, and center groups so that only three RZ parameters are used per block (rz_corner, rz_edge, rz_center) and applied both before and after the entangler. This reduces the number of trainable parameters per block from 11 to 8 (8 parameters total: 3 RZs, 3 RY, CRX ring, CRZ hub), potentially improving optimization efficiency and generalization while preserving the overall architecture (ring CRX, hub CRZ, and shared rotations). The CRX ring and hub CRZ parameters remain shared as before. This aligns with parameter-sharing strategies known to improve trainability in variational circuits. </DESCRIPTION>

def _build_hybrid_ansatz_spec() -> list[dict]:
    """Build hybrid ansatz combining parametrized ring (CRX) and shared hub (CRZ).
    
    Structure:
    1. Pre-entanglement RZ on corners/edges/center (shared per group)
    2. RY rotations on corners/edges/center (shared per group)
    3. Ring entanglement: parametrized CRX gates with shared parameter
    4. Hub entanglement: CRZ gates center-to-edges with shared parameter
    5. Post-entanglement RZ rotations with parameter sharing
    """
    spec: list[dict] = []

    # Group indices by board symmetry
    corners = (0, 2, 4, 6)
    edges = (1, 3, 5, 7)
    center = 8

    # Layer 1: Pre-entanglement RZ rotations with parameter sharing
    for w in corners:
        spec.append({"gate": "RZ", "wire": w, "param": "rz_pre_corner"})
    for w in edges:
        spec.append({"gate": "RZ", "wire": w, "param": "rz_pre_edge"})
    spec.append({"gate": "RZ", "wire": center, "param": "rz_pre_center"})

    # Layer 2: RY rotations with parameter sharing
    for w in corners:
        spec.append({"gate": "RY", "wire": w, "param": "ry_corner"})
    for w in edges:
        spec.append({"gate": "RY", "wire": w, "param": "ry_edge"})
    spec.append({"gate": "RY", "wire": center, "param": "ry_center"})

    # Layer 3: Ring entanglement (outer 8 qubits)
    # Parametrized ring entanglement from program 2
    spec.append({"gate": "CRX", "wires": [0, 1], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [1, 2], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [2, 3], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [3, 4], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [4, 5], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [5, 6], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [6, 7], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [7, 0], "param": "crx_ring"})

    # Layer 4: Hub entanglement (center to edges) with one shared parameter
    # Shared hub parameter from program 1
    spec.append({"gate": "CRY", "wires": [8, 1], "param": "cry_hub_edge"})
    spec.append({"gate": "CRY", "wires": [8, 3], "param": "cry_hub_edge"})
    spec.append({"gate": "CRY", "wires": [8, 5], "param": "cry_hub_edge"})
    spec.append({"gate": "CRY", "wires": [8, 7], "param": "cry_hub_edge"})

    # Layer 5: Post-entanglement RZ rotations with parameter sharing
    spec.append({"gate": "RZ", "wire": 0, "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wire": 2, "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wire": 4, "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wire": 6, "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wire": 1, "param": "rz_post_edge"})
    spec.append({"gate": "RZ", "wire": 3, "param": "rz_post_edge"})
    spec.append({"gate": "RZ", "wire": 5, "param": "rz_post_edge"})
    spec.append({"gate": "RZ", "wire": 7, "param": "rz_post_edge"})
    spec.append({"gate": "RZ", "wire": 8, "param": "rz_post_center"})
    
    return spec

def _build_hybrid_ansatz_spec() -> list[dict]:
    """Build hybrid ansatz combining parametrized ring (CRX) and shared hub (CRZ).
    
    Structure:
    1. Pre-entanglement RZ on corners/edges/center (shared per group)
    2. RY rotations on corners/edges/center (shared per group)
    3. Ring entanglement: parametrized CRX gates with shared parameter
    4. Hub entanglement: CRZ gates center-to-edges with shared parameter
    5. Post-entanglement RZ on corners/edges/center (shared per group)
    
    This combines:
    - Parametrized ring entanglement from program 2 (improved validation accuracy to 0.65)
    - Shared hub parameter from program 1 (reduced parameter count)
    """
    spec: list[dict] = []

    corners = (0, 2, 4, 6)
    edges = (1, 3, 5, 7)
    center = 8

    # Pre-entanglement RZ rotations (shared per group) using rz_corner/ rz_edge/ rz_center keys
    for w in corners:
        spec.append({"gate": "RZ", "wire": w, "param": "rz_corner"})
    for w in edges:
        spec.append({"gate": "RZ", "wire": w, "param": "rz_edge"})
    spec.append({"gate": "RZ", "wire": center, "param": "rz_center"})

    # Layer 2: RY rotations (shared per group)
    for w in corners:
        spec.append({"gate": "RY", "wire": w, "param": "ry_corner"})
    for w in edges:
        spec.append({"gate": "RY", "wire": w, "param": "ry_edge"})
    spec.append({"gate": "RY", "wire": center, "param": "ry_center"})

    # Layer 3: Ring entanglement (outer ring CRX with shared parameter)
    spec.append({"gate": "CRX", "wires": [0, 1], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [1, 2], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [2, 3], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [3, 4], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [4, 5], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [5, 6], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [6, 7], "param": "crx_ring"})
    spec.append({"gate": "CRX", "wires": [7, 0], "param": "crx_ring"})

    # Layer 4: Hub entanglement (center to edges) with shared parameter
    spec.append({"gate": "CRY", "wires": [8, 1], "param": "cry_hub_edge"})
    spec.append({"gate": "CRY", "wires": [8, 3], "param": "cry_hub_edge"})
    spec.append({"gate": "CRY", "wires": [8, 5], "param": "cry_hub_edge"})
    spec.append({"gate": "CRY", "wires": [8, 7], "param": "cry_hub_edge"})

    # Layer 5: Post-entanglement RZ apply using same rz keys
    for w in corners:
        spec.append({"gate": "RZ", "wire": w, "param": "rz_corner"})
    for w in edges:
        spec.append({"gate": "RZ", "wire": w, "param": "rz_edge"})
    spec.append({"gate": "RZ", "wire": center, "param": "rz_center"})

    return spec
