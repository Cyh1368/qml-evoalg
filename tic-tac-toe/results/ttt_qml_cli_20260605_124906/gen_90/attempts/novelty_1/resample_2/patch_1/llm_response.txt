<DESCRIPTION>
Inspired by the prior symmetric, parameter-shared blocks (and the potential mid-block hint that can boost expressivity), this design inserts a mid-layer of symmetry-shared rotations before the entangling stages. The ansatz structure now uses:
- Layer 1: pre-entanglement RZ rotations with symmetry-based parameter sharing (corners, edges, center)
- Layer 2: a mid-layer of RY rotations with the same symmetry groups (shared parameters)
- Layer 3: a ring of CRY gates around the outer 8 qubits (shared parameter)
- Layer 4: hub entanglement CRX gates from center to each edge (shared parameter)
- Layer 5: an additional ring-layer CRZ phase entanglement on the outer ring (shared parameter)
- Layer 6: post-entanglement RZ rotations with the same symmetry groups

This approach increases expressivity without exploding the parameter count and preserves the same data encoding, re-uploading scheme, and measurement structure. It adheres to the adjacency constraints and uses deliberate parameter sharing to improve generalization and trainability, aiming to improve validation/test performance while keeping total parameters in the same/bounded range as prior successful runs.

def _build_enhanced_ansatz_spec() -> list[dict]:
    """Enhanced symmetry-aware ansatz with a mid-block RY layer and ring/hub entanglement.

    Structure (with parameter sharing per symmetry group):
    1) Pre-entanglement RZ on all qubits, sharing within corners/edges/center
    2) Mid-layer RY on all qubits, sharing within corners/edges/center
    3) Ring entanglement: CRY around the outer ring (shared parameter)
    4) Hub entanglement: CRX from center to edges (shared parameter)
    5) Ring phase entanglement: CRZ around the outer ring (shared parameter)
    6) Post-entanglement RZ on all qubits, sharing within corners/edges/center
    """
    spec: list[dict] = []

    corners = (0, 2, 4, 6)
    edges = (1, 3, 5, 7)
    center = 8

    # Layer 1: Pre-entanglement RZ rotations (symmetry-shared)
    for w in corners:
        spec.append({"gate": "RZ", "wire": w, "param": "rz_pre_corner"})
    for w in edges:
        spec.append({"gate": "RZ", "wire": w, "param": "rz_pre_edge"})
    spec.append({"gate": "RZ", "wire": center, "param": "rz_pre_center"})

    # Layer 2: Mid-layer RY rotations (symmetry-shared)
    for w in corners:
        spec.append({"gate": "RY", "wire": w, "param": "ry_mid_corner"})
    for w in edges:
        spec.append({"gate": "RY", "wire": w, "param": "ry_mid_edge"})
    spec.append({"gate": "RY", "wire": center, "param": "ry_mid_center"})

    # Layer 3: Ring entanglement (outer ring) - CRY with shared parameter
    spec.append({"gate": "CRY", "wires": [0, 1], "param": "cry_ring"})
    spec.append({"gate": "CRY", "wires": [1, 2], "param": "cry_ring"})
    spec.append({"gate": "CRY", "wires": [2, 3], "param": "cry_ring"})
    spec.append({"gate": "CRY", "wires": [3, 4], "param": "cry_ring"})
    spec.append({"gate": "CRY", "wires": [4, 5], "param": "cry_ring"})
    spec.append({"gate": "CRY", "wires": [5, 6], "param": "cry_ring"})
    spec.append({"gate": "CRY", "wires": [6, 7], "param": "cry_ring"})
    spec.append({"gate": "CRY", "wires": [7, 0], "param": "cry_ring"})

    # Layer 4: Hub entanglement (center to edges) - CRX with shared parameter
    spec.append({"gate": "CRX", "wires": [center, 1], "param": "crx_hub_edge"})
    spec.append({"gate": "CRX", "wires": [center, 3], "param": "crx_hub_edge"})
    spec.append({"gate": "CRX", "wires": [center, 5], "param": "crx_hub_edge"})
    spec.append({"gate": "CRX", "wires": [center, 7], "param": "crx_hub_edge"})

    # Layer 5: Ring phase entanglement (outer ring) - CRZ with shared parameter
    spec.append({"gate": "CRZ", "wires": [0, 1], "param": "crz_ring"})
    spec.append({"gate": "CRZ", "wires": [1, 2], "param": "crz_ring"})
    spec.append({"gate": "CRZ", "wires": [2, 3], "param": "crz_ring"})
    spec.append({"gate": "CRZ", "wires": [3, 4], "param": "crz_ring"})
    spec.append({"gate": "CRZ", "wires": [4, 5], "param": "crz_ring"})
    spec.append({"gate": "CRZ", "wires": [5, 6], "param": "crz_ring"})
    spec.append({"gate": "CRZ", "wires": [6, 7], "param": "crz_ring"})
    spec.append({"gate": "CRZ", "wires": [7, 0], "param": "crz_ring"})

    # Layer 6: Post-entanglement RZ rotations (symmetry-shared)
    for w in corners:
        spec.append({"gate": "RZ", "wire": w, "param": "rz_post_corner"})
    for w in edges:
        spec.append({"gate": "RZ", "wire": w, "param": "rz_post_edge"})
    spec.append({"gate": "RZ", "wire": center, "param": "rz_post_center"})

    return spec


ANSATZ_SPEC = _build_enhanced_ansatz_spec()
