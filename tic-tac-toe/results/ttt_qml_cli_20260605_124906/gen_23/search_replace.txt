- Ring parametrization (from best prior)

- NEW: Hub-corner parametrization (new idea)
- Symmetry-preserving parameter sharing for efficiency

def _build_hierarchical_ansatz_spec() -> list[dict]:
    """Build hybrid hierarchical entanglement ansatz with parameter sharing.

    Structural changes (hybrid design):
    1) Pre-entanglement RZ on all qubits (shared by symmetry)
    2) RY rotations on all qubits (shared by symmetry)
    3) Ring entanglement: CZ gates around outer ring (no extra gates on ring)
    4) Hub entanglement: center-to-edges using CRZ and CRX gates
       with a single shared parameter ("hub_angle") across all hub gates
    5) Post-entanglement RZ on all qubits (shared by symmetry)
    """
    spec = []

    # Symmetry groups for 4-fold rotational symmetry of tic-tac-toe board
    # Corners: 0, 2, 4, 6 (all equivalent by rotation)
    # Edges: 1, 3, 5, 7 (all equivalent by rotation)
    # Center: 8 (unique)

    # Layer 1: Pre-entanglement RZ rotations with parameter sharing
    spec.append({"gate": "RZ", "wire": 0, "param": "rz_pre_corner"})
    spec.append({"gate": "RZ", "wire": 2, "param": "rz_pre_corner"})
    spec.append({"gate": "RZ", "wire": 4, "param": "rz_pre_corner"})
    spec.append({"gate": "RZ", "wire": 6, "param": "rz_pre_corner"})
    spec.append({"gate": "RZ", "wire": 1, "param": "rz_pre_edge"})
    spec.append({"gate": "RZ", "wire": 3, "param": "rz_pre_edge"})
    spec.append({"gate": "RZ", "wire": 5, "param": "rz_pre_edge"})
    spec.append({"gate": "RZ", "wire": 7, "param": "rz_pre_edge"})
    spec.append({"gate": "RZ", "wire": 8, "param": "rz_pre_center"})

    # Layer 2: RY rotations with parameter sharing
    spec.append({"gate": "RY", "wire": 0, "param": "ry_corner"})
    spec.append({"gate": "RY", "wire": 2, "param": "ry_corner"})
    spec.append({"gate": "RY", "wire": 4, "param": "ry_corner"})
    spec.append({"gate": "RY", "wire": 6, "param": "ry_corner"})
    spec.append({"gate": "RY", "wire": 1, "param": "ry_edge"})
    spec.append({"gate": "RY", "wire": 3, "param": "ry_edge"})
    spec.append({"gate": "RY", "wire": 5, "param": "ry_edge"})
    spec.append({"gate": "RY", "wire": 7, "param": "ry_edge"})
    spec.append({"gate": "RY", "wire": 8, "param": "ry_center"})

    # Layer 3: Ring entanglement (outer 8 qubits)
    # Keep ring entanglement as CZ-only to reduce depth and parameter count
    spec.append({"gate": "CZ", "wires": [0, 1]})
    spec.append({"gate": "CZ", "wires": [1, 2]})
    spec.append({"gate": "CZ", "wires": [2, 3]})
    spec.append({"gate": "CZ", "wires": [3, 4]})
    spec.append({"gate": "CZ", "wires": [4, 5]})
    spec.append({"gate": "CZ", "wires": [5, 6]})
    spec.append({"gate": "CZ", "wires": [6, 7]})
    spec.append({"gate": "CZ", "wires": [7, 0]})

    # Layer 4: Hub entanglement (center to edges) with a single shared parameter
    # - Center to each edge: CRZ and CRX gates share the same angle "hub_angle"
    spec.append({"gate": "CRZ", "wires": [CENTER, 1], "param": "hub_angle"})
    spec.append({"gate": "CRZ", "wires": [CENTER, 3], "param": "hub_angle"})
    spec.append({"gate": "CRZ", "wires": [CENTER, 5], "param": "hub_angle"})
    spec.append({"gate": "CRZ", "wires": [CENTER, 7], "param": "hub_angle"})

    spec.append({"gate": "CRX", "wires": [CENTER, 1], "param": "hub_angle"})
    spec.append({"gate": "CRX", "wires": [CENTER, 3], "param": "hub_angle"})
    spec.append({"gate": "CRX", "wires": [CENTER, 5], "param": "hub_angle"})
    spec.append({"gate": "CRX", "wires": [CENTER, 7], "param": "hub_angle"})

    # Layer 5: Post-entanglement RZ rotations with parameter sharing
    spec.append({"gate": "RZ", "wire": 0, "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wire": 2, "param": "rz_post_corner"})

    spec.append({"gate": "RZ", "wire": 7, "param": "rz_post_edge"})
    spec.append({"gate": "RZ", "wire": 8, "param": "rz_post_center"})

def _build_hierarchical_ansatz_spec() -> list[dict]:
    """Build enhanced hierarchical entanglement ansatz with multi-gate hub.

    Structural improvements:
    1) Pre-entanglement RZ on all qubits (shared by symmetry)
    2) RY rotations on all qubits (shared by symmetry)
    3) Ring entanglement: Parametrized CRX gates around outer ring (shared parameter)
    4) Hub entanglement:
       - Center-to-edges (CRY gates, shared parameter for edges)
       - Center-to-corners (CRY gates, shared parameter for corners)
    5) Post-entanglement RZ rotations with parameter sharing

    Rationale: Corners are special in tic-tac-toe (form winning lines and diagonals),
    so dedicated corner-to-center entanglement improves feature learning.
    """
    spec = []

    # Symmetry groups for 4-fold rotational symmetry of tic-tac-toe board
    # Corners: 0, 2, 4, 6 (all equivalent by rotation)
    # Edges: 1, 3, 5, 7 (all equivalent by rotation)
    # Center: 8 (unique)

    # Layer 1: Pre-entanglement RZ rotations with parameter sharing
    spec.append({"gate": "RZ", "wire": 0, "param": "rz_pre_corner"})
    spec.append({"gate": "RZ", "wire": 2, "param": "rz_pre_corner"})
    spec.append({"gate": "RZ", "wire": 4, "param": "rz_pre_corner"})
    spec.append({"gate": "RZ", "wire": 6, "param": "rz_pre_corner"})
    spec.append({"gate": "RZ", "wire": 1, "param": "rz_pre_edge"})
    spec.append({"gate": "RZ", "wire": 3, "param": "rz_pre_edge"})
    spec.append({"gate": "RZ", "wire": 5, "param": "rz_pre_edge"})
    spec.append({"gate": "RZ", "wire": 7, "param": "rz_pre_edge"})

    spec.append({"gate": "RY", "wire": 4, "param": "ry_corner"})

    spec.append({"gate": "CRX", "wires": [7, 0], "param": "crx_ring"})

    # Layer 4a: Hub entanglement - center to edges (CRY gates, shared parameter)
    spec.append({"gate": "CRY", "wires": [CENTER, 1], "param": "cry_hub_edge"})
    spec.append({"gate": "CRY", "wires": [CENTER, 3], "param": "cry_hub_edge"})
    spec.append({"gate": "CRY", "wires": [CENTER, 5], "param": "cry_hub_edge"})
    spec.append({"gate": "CRY", "wires": [CENTER, 7], "param": "cry_hub_edge"})

    # Layer 4b: Hub entanglement - center to corners (CRY gates, shared parameter)

    spec.append({"gate": "CRY", "wires": [CENTER, 0], "param": "cry_hub_corner"})

    # Layer 5: Post-entanglement RZ rotations with parameter sharing
    spec.append({"gate": "RZ", "wire": 0, "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wire": 2, "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wire": 4, "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wire": 6, "param": "rz_post_corner"})
    spec.append({"gate": "RZ", "wire": 1, "param": "rz_post_edge"})
    spec.append({"gate": "RZ", "wire": 3, "param": "rz_post_edge"})
    spec.append({"gate": "RZ", "wire": 5, "param": "rz_post_edge"})
