ANSATZ_SPEC = []

mirror_pairs = [(0, 7), (1, 6), (2, 5), (3, 4)]
local_pairs = [(0, 1), (2, 3), (4, 5), (6, 7)]
global_pairs = [(0, 4), (1, 5), (2, 6), (3, 7)]

# Symmetry-aware input rotation: tied parameters on mirrored qubit pairs.
for idx, (a, b) in enumerate(mirror_pairs):
    ANSATZ_SPEC.append({"gate": "RY", "wire": a, "param": f"ry_m{idx}"})
    ANSATZ_SPEC.append({"gate": "RY", "wire": b, "param": f"ry_m{idx}"})

# Phase preconditioning, also tied across mirrored pairs.
for idx, (a, b) in enumerate(mirror_pairs):
    ANSATZ_SPEC.append({"gate": "RZ", "wire": a, "param": f"rz_in{idx}"})
    ANSATZ_SPEC.append({"gate": "RZ", "wire": b, "param": f"rz_in{idx}"})

# Local butterfly mixing: parity extraction on adjacent pairs.
for a, b in local_pairs:
    ANSATZ_SPEC.append({"gate": "CNOT", "wires": [a, b]})
    ANSATZ_SPEC.append({"gate": "CRY", "wires": [a, b], "param": "mix_local"})

# Global cross-qubit mixing: long-range interactions for higher-order correlations.
for a, b in global_pairs:
    ANSATZ_SPEC.append({"gate": "CZ", "wires": [a, b]})
    ANSATZ_SPEC.append({"gate": "CRX", "wires": [a, b], "param": "mix_global"})

# Output phase refinement.
for idx, (a, b) in enumerate(mirror_pairs):
    ANSATZ_SPEC.append({"gate": "RZ", "wire": a, "param": f"rz_out{idx}"})
    ANSATZ_SPEC.append({"gate": "RZ", "wire": b, "param": f"rz_out{idx}"})

# Residual symmetry closure: reuse the same RY parameters at the end.
for idx, (a, b) in enumerate(mirror_pairs):
    ANSATZ_SPEC.append({"gate": "RY", "wire": a, "param": f"ry_m{idx}"})
    ANSATZ_SPEC.append({"gate": "RY", "wire": b, "param": f"ry_m{idx}"})
# EVOLVE-BLOCK-END
