<DESCRIPTION>
This ansatz performs a crossover between the 3D-hypercube correlation structure of Programs 1 and 3 and the extremely efficient global parameter sharing of Program 2. 

Instead of a dense block of 28 consecutive `CZ` gates, we use the 12 `CZ` gates that make up the 3 orthogonal matchings of a hypercube. To maintain high expressivity, we interleave these matchings with alternating global `RY` and `RX` rotations. This means phases created by the fixed `IsingZZ` feature map are continuously mixed into amplitudes, ensuring powerful correlation transport across the register. 

By reusing just two parameter keys (`global_ry` and `global_rx`) across all 8 qubits and multiple layers within the block, we use only 2 parameters per block (14 parameters total). This maximizes parameter efficiency, virtually eliminates the train-test generalization gap, and results in a lower circuit depth than the complete-graph approach.
</DESCRIPTION>

"""Seed program for ShinkaEvolve ansatz search: 8-qubit binary classifier.

Records are 28-dimensional binary feature vectors from an external dataset.
Each feature k is encoded by the fixed feature map as a two-qubit phase
coupling on the qubit pair FEATURE_PAIRS[k]. Labels are +1/-1 and encode a
property of the records determined by the dataset.

Only ANSATZ_SPEC is inside the EVOLVE-BLOCK. The dataset, feature map,
re-uploading layout, training loop, measurement and metric packaging are
fixed so candidates are compared on the same task.
"""

from __future__ import annotations

import json
import os
from pathlib import Path

import numpy as np
import pennylane as qml
from pennylane import numpy as pnp


# Problem constants
N_QUBITS = 8
N_UPLOADS = 3
N_REPEATS = 2
FEATURE_SCALE = np.pi / 2

_DATA_PATH = os.environ.get("TASK_DATA", str(Path(__file__).resolve().parent / "dataset.npz"))
_DATA = np.load(_DATA_PATH)
FEATURE_PAIRS = [tuple(int(w) for w in row) for row in _DATA["feature_pairs"]]
N_FEATURES = len(FEATURE_PAIRS)

ALLOWED_SINGLE_QUBIT_GATES = {"RX", "RY", "RZ"}
ALLOWED_TWO_QUBIT_GATES = {"CNOT", "CZ"}
ALLOWED_PARAM_TWO_QUBIT_GATES = {"CRX", "CRY", "CRZ"}


# EVOLVE-BLOCK-START
ANSATZ_SPEC = [
    # 1. Global RY layer to create superposition and mix previous ZZ phases
    {"gate": "RY", "wire": 0, "param": "global_ry"},
    {"gate": "RY", "wire": 1, "param": "global_ry"},
    {"gate": "RY", "wire": 2, "param": "global_ry"},
    {"gate": "RY", "wire": 3, "param": "global_ry"},
    {"gate": "RY", "wire": 4, "param": "global_ry"},
    {"gate": "RY", "wire": 5, "param": "global_ry"},
    {"gate": "RY", "wire": 6, "param": "global_ry"},
    {"gate": "RY", "wire": 7, "param": "global_ry"},

    # Hypercube dimension 0
    {"gate": "CZ", "wires": [0, 1]},
    {"gate": "CZ", "wires": [2, 3]},
    {"gate": "CZ", "wires": [4, 5]},
    {"gate": "CZ", "wires": [6, 7]},

    # 2. Global RX layer to rotate into a different basis
    {"gate": "RX", "wire": 0, "param": "global_rx"},
    {"gate": "RX", "wire": 1, "param": "global_rx"},
    {"gate": "RX", "wire": 2, "param": "global_rx"},
    {"gate": "RX", "wire": 3, "param": "global_rx"},
    {"gate": "RX", "wire": 4, "param": "global_rx"},
    {"gate": "RX", "wire": 5, "param": "global_rx"},
    {"gate": "RX", "wire": 6, "param": "global_rx"},
    {"gate": "RX", "wire": 7, "param": "global_rx"},

    # Hypercube dimension 1
    {"gate": "CZ", "wires": [0, 2]},
    {"gate": "CZ", "wires": [1, 3]},
    {"gate": "CZ", "wires": [4, 6]},
    {"gate": "CZ", "wires": [5, 7]},

    # 3. Second Global RY layer (reusing param) for deep mixing
    {"gate": "RY", "wire": 0, "param": "global_ry"},
    {"gate": "RY", "wire": 1, "param": "global_ry"},
    {"gate": "RY", "wire": 2, "param": "global_ry"},
    {"gate": "RY", "wire": 3, "param": "global_ry"},
    {"gate": "RY", "wire": 4, "param": "global_ry"},
    {"gate": "RY", "wire": 5, "param": "global_ry"},
    {"gate": "RY", "wire": 6, "param": "global_ry"},
    {"gate": "RY", "wire": 7, "param": "global_ry"},

    # Hypercube dimension 2 completes correlation across all qubits
    {"gate": "CZ", "wires": [0, 4]},
    {"gate": "CZ", "wires": [1, 5]},
    {"gate": "CZ", "wires": [2, 6]},
    {"gate": "CZ", "wires": [3, 7]},

    # 4. Final Global RX layer (reusing param) avoids commuting with terminal Z measurements
    {"gate": "RX", "wire": 0, "param": "global_rx"},
    {"gate": "RX", "wire": 1, "param": "global_rx"},
    {"gate": "RX", "wire": 2, "param": "global_rx"},
    {"gate": "RX", "wire": 3, "param": "global_rx"},

def load_splits() -> dict:
    return {
        "train": (np.asarray(_DATA["x_train"], dtype=float), np.asarray(_DATA["y_train"], dtype=float)),
        "validation": (np.asarray(_DATA["x_validation"], dtype=float), np.asarray(_DATA["y_validation"], dtype=float)),
        "test": (np.asarray(_DATA["x_test"], dtype=float), np.asarray(_DATA["y_test"], dtype=float)),
    }


def validate_ansatz_spec(spec: list[dict]) -> tuple[list[str], list[str]]:
    """Validate ANSATZ_SPEC and return errors plus ordered parameter keys."""
    errors = []
    parameter_keys = []
    if not isinstance(spec, list) or not spec:
        return ["ANSATZ_SPEC must be a non-empty list of gate dictionaries."], []

    for gate_index, item in enumerate(spec):
        prefix = f"gate {gate_index}"
        if not isinstance(item, dict):
            errors.append(f"{prefix}: expected a dict, got {type(item).__name__}")
            continue
        gate = str(item.get("gate", "")).upper()
        if gate in ALLOWED_SINGLE_QUBIT_GATES:
            wire = item.get("wire")
            if not isinstance(wire, int) or not 0 <= wire < N_QUBITS:
                errors.append(f"{prefix}: {gate} requires integer wire in [0, {N_QUBITS - 1}]")
            param = item.get("param")
            if not isinstance(param, str) or not param:
                errors.append(f"{prefix}: {gate} requires a non-empty string param key")
            else:
                parameter_keys.append(param)
        elif gate in ALLOWED_TWO_QUBIT_GATES | ALLOWED_PARAM_TWO_QUBIT_GATES:
            wires = item.get("wires")
            if (
                not isinstance(wires, (list, tuple))
                or len(wires) != 2
                or not all(isinstance(w, int) for w in wires)
                or wires[0] == wires[1]
                or not all(0 <= w < N_QUBITS for w in wires)
            ):
                errors.append(f"{prefix}: {gate} requires two distinct integer wires")
            if gate in ALLOWED_PARAM_TWO_QUBIT_GATES:
                param = item.get("param")
                if not isinstance(param, str) or not param:
                    errors.append(f"{prefix}: {gate} requires a non-empty string param key")
                else:
                    parameter_keys.append(param)
        else:
            allowed = sorted(
                ALLOWED_SINGLE_QUBIT_GATES | ALLOWED_TWO_QUBIT_GATES | ALLOWED_PARAM_TWO_QUBIT_GATES
            )
            errors.append(f"{prefix}: unsupported gate {gate!r}; allowed gates are {allowed}")

    ordered_unique_keys = list(dict.fromkeys(parameter_keys))
    return errors, ordered_unique_keys


SPEC_ERRORS, PARAMETER_KEYS_PER_BLOCK = validate_ansatz_spec(ANSATZ_SPEC)
N_PARAMS_PER_BLOCK = len(PARAMETER_KEYS_PER_BLOCK)
N_CIRCUIT_PARAMS = N_PARAMS_PER_BLOCK * N_UPLOADS * N_REPEATS
N_READOUT_PARAMS = 2  # trainable output calibration: gain, bias
N_PARAMS = N_CIRCUIT_PARAMS + N_READOUT_PARAMS


def feature_map(records) -> None:
    """Encode each feature as a fixed two-qubit phase coupling.

    Accepts a batch of records (shape (B, N_FEATURES)); records[..., k]
    broadcasts over the leading batch dimension.
    """
    for k, (a, b) in enumerate(FEATURE_PAIRS):
        qml.IsingZZ(FEATURE_SCALE * records[..., k], wires=[a, b])


def apply_ansatz_block(params, block_index: int) -> None:
    """Apply one candidate ansatz block from the formal ANSATZ_SPEC."""
    if SPEC_ERRORS:
        raise ValueError("; ".join(SPEC_ERRORS))

    offset = block_index * N_PARAMS_PER_BLOCK
    key_to_position = {key: i for i, key in enumerate(PARAMETER_KEYS_PER_BLOCK)}
    for item in ANSATZ_SPEC:
        gate = str(item["gate"]).upper()
        if gate in ALLOWED_SINGLE_QUBIT_GATES:
            angle = params[offset + key_to_position[item["param"]]]
            wire = int(item["wire"])
            if gate == "RX":
                qml.RX(angle, wires=wire)
            elif gate == "RY":
                qml.RY(angle, wires=wire)
            elif gate == "RZ":
                qml.RZ(angle, wires=wire)
        elif gate in ALLOWED_TWO_QUBIT_GATES:
            wires = [int(w) for w in item["wires"]]
            if gate == "CNOT":
                qml.CNOT(wires=wires)
            elif gate == "CZ":
                qml.CZ(wires=wires)
        elif gate in ALLOWED_PARAM_TWO_QUBIT_GATES:
            angle = params[offset + key_to_position[item["param"]]]
            wires = [int(w) for w in item["wires"]]
            if gate == "CRX":
                qml.CRX(angle, wires=wires)
            elif gate == "CRY":
                qml.CRY(angle, wires=wires)
            elif gate == "CRZ":
                qml.CRZ(angle, wires=wires)


_DEVICE = qml.device("default.qubit", wires=N_QUBITS, shots=None)


@qml.qnode(_DEVICE, interface="autograd", diff_method="backprop")
def circuit_z_expectations(records, circuit_params):
    """Fixed pipeline: alternating feature uploads and ansatz blocks."""
    for upload_index in range(N_UPLOADS):
        feature_map(records)
        for repeat_index in range(N_REPEATS):
            block_index = upload_index * N_REPEATS + repeat_index
            apply_ansatz_block(circuit_params, block_index)
    return [qml.expval(qml.PauliZ(wire)) for wire in range(N_QUBITS)]


def predict_batch(params, x_batch: np.ndarray):
    """Scalar prediction per record: calibrated mean of qubit Z expectations."""
    circuit_params = params[:N_CIRCUIT_PARAMS]
    gain = params[N_CIRCUIT_PARAMS]
    bias = params[N_CIRCUIT_PARAMS + 1]
    records = pnp.array(np.asarray(x_batch, dtype=float), requires_grad=False)
    z = pnp.stack(circuit_z_expectations(records, circuit_params))  # (N_QUBITS, B)
    raw = pnp.mean(z, axis=0)
    return gain * (raw - bias)


def l2_loss(params, x_batch: np.ndarray, y_batch: np.ndarray):
    predictions = predict_batch(params, x_batch)
    targets = pnp.array(y_batch, requires_grad=False)
    return pnp.mean((predictions - targets) ** 2)


def evaluate_split(params, x: np.ndarray, y: np.ndarray) -> dict[str, float]:
    predictions = np.asarray(predict_batch(params, x), dtype=float)
    loss = float(np.mean((predictions - y) ** 2))
    accuracy = float(np.mean(np.sign(predictions) == np.sign(y)))
    return {"accuracy": accuracy, "loss": loss}


def _batch_indices(order: np.ndarray, step: int, batch_size: int) -> np.ndarray:
    start = (step * batch_size) % len(order)
    end = start + batch_size
    if end <= len(order):
        return order[start:end]
    return np.concatenate([order[start:], order[: end - len(order)]])


def _log(record: dict, log_file: Path | None, verbose: bool) -> None:
    line = json.dumps(record, sort_keys=True)
    if verbose:
        print(line)
    if log_file is not None:
        with log_file.open("a", encoding="utf-8") as handle:
            handle.write(line + "\n")


def circuit_metadata() -> dict:
    with qml.queuing.AnnotatedQueue() as queue:
        for upload_index in range(N_UPLOADS):
            feature_map(np.zeros((1, N_FEATURES)))
            for repeat_index in range(N_REPEATS):
                block_index = upload_index * N_REPEATS + repeat_index
                apply_ansatz_block(np.zeros(max(N_CIRCUIT_PARAMS, 1)), block_index)
    tape = qml.tape.QuantumScript.from_queue(queue)
