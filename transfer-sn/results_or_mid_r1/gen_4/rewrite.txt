"""Modular ansatz search for 8-qubit binary classifier with improved architecture.

Restructures the program using object-oriented design for better separation of concerns,
parameter management, and circuit organization. Maintains the same interface while
improving modularity and performance.
"""

from __future__ import annotations

import json
import os
from pathlib import Path
from typing import Any, NamedTuple

import numpy as np
import pennylane as qml
from pennylane import numpy as pnp


# Problem constants
N_QUBITS = 8
N_UPLOADS = 3
N_REPEATS = 2
FEATURE_SCALE = np.pi / 2

_DATA_PATH = os.environ.get("TASK_DATA", str(Path(__file__).resolve().parent / "dataset.npz"))
_DATA = np.load(_DATA_PATH)
FEATURE_PAIRS = [tuple(int(w) for w in row) for row in _DATA["feature_pairs"]]
N_FEATURES = len(FEATURE_PAIRS)

ALLOWED_SINGLE_QUBIT_GATES = {"RX", "RY", "RZ"}
ALLOWED_TWO_QUBIT_GATES = {"CNOT", "CZ"}
ALLOWED_PARAM_TWO_QUBIT_GATES = {"CRX", "CRY", "CRZ"}


class GateSpec(NamedTuple):
    """Specification for a single gate in the ansatz."""
    gate: str
    wire: int | None = None
    wires: list[int] | None = None
    param: str | None = None


class CircuitManager:
    """Manages ansatz specification, validation, and parameter mapping."""
    
    def __init__(self, ansatz_spec: list[dict]):
        self.spec = ansatz_spec
        self.errors: list[str] = []
        self.param_keys: list[str] = []
        self._validate()
    
    def _validate(self) -> None:
        """Validate ansatz specification."""
        if not isinstance(self.spec, list) or not self.spec:
            self.errors.append("ANSATZ_SPEC must be a non-empty list of gate dictionaries.")
            return
        
        for gate_index, item in enumerate(self.spec):
            prefix = f"gate {gate_index}"
            if not isinstance(item, dict):
                self.errors.append(f"{prefix}: expected a dict, got {type(item).__name__}")
                continue
            
            gate = str(item.get("gate", "")).upper()
            
            if gate in ALLOWED_SINGLE_QUBIT_GATES:
                wire = item.get("wire")
                if not isinstance(wire, int) or not 0 <= wire < N_QUBITS:
                    self.errors.append(f"{prefix}: {gate} requires integer wire in [0, {N_QUBITS - 1}]")
                param = item.get("param")
                if not isinstance(param, str) or not param:
                    self.errors.append(f"{prefix}: {gate} requires a non-empty string param key")
                else:
                    self.param_keys.append(param)
            
            elif gate in ALLOWED_TWO_QUBIT_GATES | ALLOWED_PARAM_TWO_QUBIT_GATES:
                wires = item.get("wires")
                if (not isinstance(wires, (list, tuple)) or len(wires) != 2 or
                    not all(isinstance(w, int) for w in wires) or wires[0] == wires[1] or
                    not all(0 <= w < N_QUBITS for w in wires)):
                    self.errors.append(f"{prefix}: {gate} requires two distinct integer wires")
                
                if gate in ALLOWED_PARAM_TWO_QUBIT_GATES:
                    param = item.get("param")
                    if not isinstance(param, str) or not param:
                        self.errors.append(f"{prefix}: {gate} requires a non-empty string param key")
                    else:
                        self.param_keys.append(param)
            else:
                allowed = sorted(ALLOWED_SINGLE_QUBIT_GATES | ALLOWED_TWO_QUBIT_GATES | 
                                ALLOWED_PARAM_TWO_QUBIT_GATES)
                self.errors.append(f"{prefix}: unsupported gate {gate!r}; allowed: {allowed}")
    
    @property
    def n_params_per_block(self) -> int:
        """Number of unique parameters per block."""
        return len(list(dict.fromkeys(self.param_keys)))
    
    @property
    def is_valid(self) -> bool:
        """Check if specification is valid."""
        return len(self.errors) == 0
    
    def apply_block(self, params: pnp.ndarray, block_index: int) -> None:
        """Apply one ansatz block using the given parameters."""
        if not self.is_valid:
            raise ValueError("; ".join(self.errors))
        
        unique_keys = list(dict.fromkeys(self.param_keys))
        offset = block_index * len(unique_keys)
        key_to_pos = {key: i for i, key in enumerate(unique_keys)}
        
        for item in self.spec:
            gate = str(item["gate"]).upper()
            
            if gate in ALLOWED_SINGLE_QUBIT_GATES:
                angle = params[offset + key_to_pos[item["param"]]]
                wire = int(item["wire"])
                getattr(qml, gate)(angle, wires=wire)
            
            elif gate in ALLOWED_TWO_QUBIT_GATES:
                wires = [int(w) for w in item["wires"]]
                getattr(qml, gate)(wires=wires)
            
            elif gate in ALLOWED_PARAM_TWO_QUBIT_GATES:
                angle = params[offset + key_to_pos[item["param"]]]
                wires = [int(w) for w in item["wires"]]
                getattr(qml, gate)(angle, wires=wires)

ANSATZ_SPEC = [
    # Parametrized rotations on all qubits
    {"gate": "RY", "wire": 0, "param": "ry_a"},
    {"gate": "RY", "wire": 1, "param": "ry_a"},
    {"gate": "RY", "wire": 2, "param": "ry_a"},
    {"gate": "RY", "wire": 3, "param": "ry_a"},
    {"gate": "RY", "wire": 4, "param": "ry_b"},
    {"gate": "RY", "wire": 5, "param": "ry_b"},
    {"gate": "RY", "wire": 6, "param": "ry_b"},
    {"gate": "RY", "wire": 7, "param": "ry_b"},
    # Ring connectivity with controlled rotations
    {"gate": "CRX", "wires": [0, 1], "param": "crx_0"},
    {"gate": "CRX", "wires": [1, 2], "param": "crx_1"},
    {"gate": "CRX", "wires": [2, 3], "param": "crx_2"},

    {"gate": "RZ", "wire": 0, "param": "rz_c"},

]

class QuantumClassifier:
    """Wraps quantum circuit and classical readout for binary classification."""
    
    def __init__(self, circuit_manager: CircuitManager, device: qml.Device):
        self.manager = circuit_manager
        self.device = device
        self.n_circuit_params = (circuit_manager.n_params_per_block * N_UPLOADS * N_REPEATS 
                                if circuit_manager.is_valid else 0)
        self.n_readout_params = 2
        self.n_total_params = self.n_circuit_params + self.n_readout_params
    
    def _build_circuit(self, records, circuit_params):
        """Build and execute the quantum circuit."""
        for upload_idx in range(N_UPLOADS):
            # Feature encoding
            for k, (a, b) in enumerate(FEATURE_PAIRS):
                qml.IsingZZ(FEATURE_SCALE * records[..., k], wires=[a, b])
            # Ansatz blocks
            for repeat_idx in range(N_REPEATS):
                block_idx = upload_idx * N_REPEATS + repeat_idx
                self.manager.apply_block(circuit_params, block_idx)
        
        return [qml.expval(qml.PauliZ(w)) for w in range(N_QUBITS)]
    
    def _create_qnode(self):
        """Create QNode for circuit evaluation."""
        @qml.qnode(self.device, interface="autograd", diff_method="backprop")
        def qnode(records, circuit_params):
            return self._build_circuit(records, circuit_params)
        return qnode
    
    def predict(self, params: pnp.ndarray, x_batch: np.ndarray) -> pnp.ndarray:
        """Predict on batch of records."""
        circuit_params = params[:self.n_circuit_params]
        gain = params[self.n_circuit_params]
        bias = params[self.n_circuit_params + 1]
        
        records = pnp.array(np.asarray(x_batch, dtype=float), requires_grad=False)
        qnode = self._create_qnode()
        z_expectations = pnp.stack(qnode(records, circuit_params))
        raw_mean = pnp.mean(z_expectations, axis=0)
        
        return gain * (raw_mean - bias)
    
    def loss(self, params: pnp.ndarray, x_batch: np.ndarray, y_batch: np.ndarray) -> pnp.ndarray:
        """Compute L2 loss."""
        predictions = self.predict(params, x_batch)
        targets = pnp.array(y_batch, requires_grad=False)
        return pnp.mean((predictions - targets) ** 2)
    
    def evaluate(self, params: np.ndarray, x: np.ndarray, y: np.ndarray) -> dict[str, float]:
        """Evaluate on a split."""
        predictions = np.asarray(self.predict(params, x), dtype=float)
        loss = float(np.mean((predictions - y) ** 2))
        accuracy = float(np.mean(np.sign(predictions) == np.sign(y)))
        return {"accuracy": accuracy, "loss": loss}


def load_splits() -> dict:
    """Load dataset splits."""
    return {
        "train": (np.asarray(_DATA["x_train"], dtype=float), 
                  np.asarray(_DATA["y_train"], dtype=float)),
        "validation": (np.asarray(_DATA["x_validation"], dtype=float), 
                      np.asarray(_DATA["y_validation"], dtype=float)),
        "test": (np.asarray(_DATA["x_test"], dtype=float), 
                np.asarray(_DATA["y_test"], dtype=float)),
    }


def circuit_metadata(manager: CircuitManager) -> dict:
    """Extract circuit metadata."""
    with qml.queuing.AnnotatedQueue() as queue:
        for _ in range(N_UPLOADS):
            for k, (a, b) in enumerate(FEATURE_PAIRS):
                qml.IsingZZ(FEATURE_SCALE * 0, wires=[a, b])
            for repeat_idx in range(N_REPEATS):
                manager.apply_block(np.zeros(max(manager.n_params_per_block, 1)), 
                                   repeat_idx)
    
    tape = qml.tape.QuantumScript.from_queue(queue)
    operations = [(op.name, op.wires.tolist()) for op in tape.operations]
    depth = tape.graph.get_depth() if hasattr(tape.graph, "get_depth") else len(operations)
    
    return {
        "operations": operations,
        "depth": int(depth),
        "gate_count": int(len(operations)),
    }


def _batch_indices(order: np.ndarray, step: int, batch_size: int) -> np.ndarray:
    """Get batch indices for current step."""
    start = (step * batch_size) % len(order)
    end = start + batch_size
    if end <= len(order):
        return order[start:end]
    return np.concatenate([order[start:], order[: end - len(order)]])


def _log(record: dict, log_file: Path | None, verbose: bool) -> None:
    """Log record to file and optionally stdout."""
    line = json.dumps(record, sort_keys=True)
    if verbose:
        print(line)
    if log_file is not None:
        with log_file.open("a", encoding="utf-8") as handle:
            handle.write(line + "\n")


def run_experiment(
    seed: int = 0,
    n_epochs: int | None = None,
    steps_per_epoch: int | None = None,
    batch_size: int | None = None,
    learning_rate: float | None = None,
    convergence_threshold: float | None = None,
    eval_every_epochs: int | None = None,
    verbose: bool | None = None,
    **_ignored,
) -> dict:
    """Train and evaluate the classifier."""
    # Initialize circuit manager
    manager = CircuitManager(ANSATZ_SPEC)
    
    if not manager.is_valid:
        return {
            "spec_valid": False,
            "error": "; ".join(manager.errors),
            "n_qubits": N_QUBITS,
            "n_uploads": N_UPLOADS,
            "n_repeats": N_REPEATS,
            "n_params": 0,
            "operations": [],
            "train_accuracy": 0.0,
            "validation_accuracy": 0.0,
            "test_accuracy": 0.0,
            "train_loss": float("inf"),
            "validation_loss": float("inf"),
            "test_loss": float("inf"),
        }
    
    # Parse hyperparameters
    n_epochs = int(os.environ.get("N_EPOCHS", "1000") if n_epochs is None else n_epochs)
    steps_per_epoch = int(os.environ.get("STEPS_PER_EPOCH", "30") if steps_per_epoch is None else steps_per_epoch)
    batch_size = int(os.environ.get("BATCH_SIZE", "15") if batch_size is None else batch_size)
    learning_rate = float(os.environ.get("LEARNING_RATE", "0.03") if learning_rate is None else learning_rate)
    convergence_threshold = float(os.environ.get("CONVERGENCE_THRESHOLD", "0.90") 
                                 if convergence_threshold is None else convergence_threshold)
    eval_every_epochs = int(os.environ.get("EVAL_EVERY_EPOCHS", "1") if eval_every_epochs is None else eval_every_epochs)
    verbose = bool(int(os.environ.get("VERBOSE_TRAINING", "0"))) if verbose is None else bool(verbose)
    early_stopping = bool(int(os.environ.get("EARLY_STOPPING", "1")))
    patience = int(os.environ.get("PATIENCE", "75"))
    min_delta = float(os.environ.get("MIN_DELTA", "1e-4"))
    
    # Load data
    splits = load_splits()
    x_train, y_train = splits["train"]
    x_validation, y_validation = splits["validation"]
    x_test, y_test = splits["test"]
    
    # Setup logging
    log_file = None
    log_root = os.environ.get("TTT_LOG_DIR")
    if log_root:
        Path(log_root).mkdir(parents=True, exist_ok=True)
        log_file = Path(log_root) / f"candidate_seed_{seed}.jsonl"
        if log_file.exists():
            log_file.unlink()
    
    # Initialize classifier and parameters
    device = qml.device("default.qubit", wires=N_QUBITS, shots=None)
    classifier = QuantumClassifier(manager, device)
    
    rng = np.random.default_rng(seed)
    init_params = rng.uniform(-0.05, 0.05, size=classifier.n_total_params)
    init_params[classifier.n_circuit_params] = 1.0
    init_params[classifier.n_circuit_params + 1] = 0.0
    params = pnp.array(init_params, requires_grad=True)
    
    # Setup optimizer
    optimizer = qml.AdamOptimizer(stepsize=learning_rate)
    max_steps = n_epochs * steps_per_epoch
    
    # Tracking variables
    global_step = 0
    convergence_epoch = None
    convergence_step = None
    history = []
    best_val_loss = float("inf")
    best_params = params.copy()
    epochs_no_improve = 0
    stop_reason = "max_epochs"
    
    _log(
        {
            "event": "start",
            "seed": int(seed),
            "n_params": int(classifier.n_total_params),
            "n_params_per_block": int(manager.n_params_per_block),
            "n_epochs": int(n_epochs),
            "steps_per_epoch": int(steps_per_epoch),
            "batch_size": int(batch_size),
            "learning_rate": float(learning_rate),
        },
        log_file,
        verbose,
    )
    
    # Training loop
    for epoch in range(1, n_epochs + 1):
        order = rng.permutation(len(x_train))
        last_batch_loss = None
        
        for step in range(steps_per_epoch):
            batch_ids = _batch_indices(order, step, batch_size)
            batch_x = x_train[batch_ids]
            batch_y = y_train[batch_ids]
            
            params, last_batch_loss = optimizer.step_and_cost(
                lambda p: classifier.loss(p, batch_x, batch_y),
                params,
            )
            global_step += 1
        
        # Periodic evaluation
        if epoch == 1 or epoch == n_epochs or epoch % eval_every_epochs == 0:
            val_metrics = classifier.evaluate(params, x_validation, y_validation)
            
            if convergence_epoch is None and val_metrics["accuracy"] >= convergence_threshold:
                convergence_epoch = epoch
                convergence_step = global_step
            
            record = {
                "event": "epoch",
                "epoch": int(epoch),
                "global_step": int(global_step),
                "batch_loss": float(last_batch_loss),
                "validation_accuracy": val_metrics["accuracy"],
                "validation_loss": val_metrics["loss"],
            }
            history.append(record)
            _log(record, log_file, verbose)
            
            # Early stopping with best-weight restoration
            if val_metrics["loss"] < best_val_loss - min_delta:
                best_val_loss = val_metrics["loss"]
                best_params = params.copy()
                epochs_no_improve = 0
            else:
                epochs_no_improve += 1
                if early_stopping and epochs_no_improve >= patience:
                    stop_reason = "early_stopping"
                    break
    
    # Final evaluation
    params = best_params
    final_train = classifier.evaluate(params, x_train, y_train)
    final_validation = classifier.evaluate(params, x_validation, y_validation)
    final_test = classifier.evaluate(params, x_test, y_test)
    metadata = circuit_metadata(manager)
    
    generalization_gap = abs(final_train["accuracy"] - final_test["accuracy"])
    parameter_efficiency = final_test["accuracy"] / max(float(classifier.n_total_params), 1.0)
    
    result = {
        "spec_valid": True,
        "n_qubits": N_QUBITS,
        "n_uploads": N_UPLOADS,
        "n_repeats": N_REPEATS,
        "n_params_per_block": manager.n_params_per_block,
        "n_params": classifier.n_total_params,
        "depth": metadata["depth"],
        "gate_count": metadata["gate_count"],
        "operations": metadata["operations"],
        "train_accuracy": final_train["accuracy"],
        "validation_accuracy": final_validation["accuracy"],
        "test_accuracy": final_test["accuracy"],
        "train_loss": final_train["loss"],
        "validation_loss": final_validation["loss"],
        "test_loss": final_test["loss"],
        "generalization_gap": generalization_gap,
        "parameter_efficiency": parameter_efficiency,
        "convergence_threshold": convergence_threshold,
        "convergence_epoch": convergence_epoch,
        "convergence_step": convergence_step,
        "max_steps": max_steps,
        "stop_reason": stop_reason,
        "restore_best_weights": True,
        "history_tail": history[-10:],
    }
    
    _log({"event": "final", **result}, log_file, verbose)
    return result
