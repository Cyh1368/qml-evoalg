ANSATZ_SPEC = []

# Input mixer layer, split by bipartite parity
for w in range(8):
    parity = bin(w).count('1') % 2
    ANSATZ_SPEC.append({"gate": "RY", "wire": w, "param": f"bipartite_ry_in_{parity}"})

# Trainable phase, split by bipartite parity
for w in range(8):
    parity = bin(w).count('1') % 2
    ANSATZ_SPEC.append({"gate": "RZ", "wire": w, "param": f"bipartite_rz_{parity}"})

# Hypercube entangler (connects parity 0 to parity 1 exactly)
for i in [0, 2, 4, 6]:
    ANSATZ_SPEC.append({"gate": "CZ", "wires": [i, i + 1]})
for i in [0, 1, 4, 5]:
    ANSATZ_SPEC.append({"gate": "CZ", "wires": [i, i + 2]})
for i in [0, 1, 2, 3]:
    ANSATZ_SPEC.append({"gate": "CZ", "wires": [i, i + 4]})

# Output RY mixer, split by bipartite parity
for w in range(8):
    parity = bin(w).count('1') % 2
    ANSATZ_SPEC.append({"gate": "RY", "wire": w, "param": f"bipartite_ry_out_{parity}"})

# Output RX mixer, split by bipartite parity
for w in range(8):
